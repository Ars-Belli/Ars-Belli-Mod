# Runs monthly in country scope. Each Location has exactly one owner, so iterating
# every owner's locations visits each Location once per month.
#
# FOUNDATION TEST: set the cached count to 99 on every owned Location that has a
# trade office. The allow gate reads var:ab_to_count. If, after a month, building a
# trade office is blocked in a Location that has one, then: the pulse fires, set_variable
# works in effect context, and the location variable is readable in the build allow.
ab_trade_office_recount = {
	every_owned_location = {
		limit = { has_building = building_type:trade_office }
		set_variable = { name = ab_to_count value = 99 }
	}
}

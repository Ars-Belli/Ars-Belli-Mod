# Medium-value raw goods (vanilla default price 1.5 - 2, not food staples). Everything not in
# a list here is low value: clay, sand (0.5), stone, medicaments (1).
abm_has_medium_value_raw_material = {
	OR = {
		raw_material = goods:lumber			# 1.5
		raw_material = goods:coal			# 2
		raw_material = goods:tin			# 2
		raw_material = goods:lead			# 2
		raw_material = goods:fiber_crops	# 2
		raw_material = goods:saltpeter		# 2
		raw_material = goods:wine			# 2
		raw_material = goods:beeswax		# 2, food 2.5
		raw_material = goods:fur			# 2, food 2
	}
}

# Food tiers by vanilla food value. A town loses the rural food bonus and makes 10% less food,
# a city 20% less, so the more food a location makes, the later it should urbanise.
abm_has_food_8_raw_material = {
	OR = {
		raw_material = goods:rice			# 10
		raw_material = goods:wheat			# 8
		raw_material = goods:maize			# 8
		raw_material = goods:potato			# 8
		raw_material = goods:livestock		# 8
	}
}

abm_has_food_5_raw_material = {

# Medium-value goods in a rich location: a town halves its RGO size (loses the rural +100%),
# which costs more than the cheap goods suggest.
abm_has_rich_medium_value_rgo = {
	abm_has_medium_value_raw_material = yes
	location_tax_base >= abm_town_rich_rgo_tax_base
}

# Where a bailiff could stand (vanilla bailiff: rural_settlement, not overseas), limited to
# integrated or cored locations.
abm_bailiff_location_eligible = {
	location_rank = location_rank:rural_settlement
	is_overseas_for_owner = no
	OR = {
		integration_level = integrated
		integration_level = core
	}
}

# A bailiff (+20% local max control) pays for its upkeep, or the goods are worth growing into.
abm_bailiff_worth_it = {
	OR = {
		location_tax_base >= abm_bailiff_min_tax_base
		abm_has_high_value_raw_material = yes
	}
}

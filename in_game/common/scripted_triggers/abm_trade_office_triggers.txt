# Ars Belli trade office limit. Location scope, scope:actor = the country building.
#
# Why a cached count: inside a building's allow, the location building iterators
# (any_foreign_buildings_in_location and friends) come back empty, and a global any_country
# per candidate location lags the build menu. So the number of trade offices in a location is
# kept in the location variable abm_trade_office_count, recounted in effect context (where the
# iterators work) whenever a trade office is started, finished, cancelled or destroyed, at game
# start and yearly - see scripted_effects/abm_trade_office_effects.txt. Here it is only read.
#
# Called from the allow of REPLACE:trade_office (building_types/abm_foreign_buildings.txt).
abm_trade_office_slot_free = {
	custom_tooltip = {
		text = abm_trade_office_slot_free_tt
		OR = {
			# upgrading an office you already have here never takes a new slot
			location_building_level = {
				building_type = building_type:trade_office
				owner = scope:actor
				value > 0
			}
			# never counted: no trade office has been built or started here yet
			NOT = { has_variable = abm_trade_office_count }
			var:abm_trade_office_count < abm_trade_office_limit_value
		}
	}
}

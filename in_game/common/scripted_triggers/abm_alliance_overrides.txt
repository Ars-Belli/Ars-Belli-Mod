REPLACE:alliance_can_exist = {
	# Guard: never evaluate a country against itself
	$first$ != $second$
	$first$ = {
		opinion = { target = $second$ value >= 0 }
		not = { has_blocked_treaties = $second$ }
		not = { is_subject_of = $second$ }
		not = { is_rival_of = $second$ }
		modifier:allowed_alliance = yes
		NOT = {
			is_member_of_international_organization_of_type = {
				type = coalition
				target = $second$
			}
		}
		NOT = {
			is_member_of_international_organization_of_type = {
				type = abm_player_coalition
				target = $second$
			}
		}
		NOT = {
			in_union_with = $second$
		}
		is_rebel_country = no
		NOT = {
			AND = {
				is_mp_gp = yes
				$second$ = { is_mp_gp = yes }
			}
		}
	}
	$second$ = {
		opinion = { target = $first$ value >= 0 }
		not = { has_blocked_treaties = $first$ }
		not = { is_subject_of = $first$ }
		not = { is_rival_of = $first$ }
		modifier:allowed_alliance = yes
		NOT = {
			is_member_of_international_organization_of_type = {
				type = coalition
				target = $first$
			}
		}
		NOT = {
			is_member_of_international_organization_of_type = {
				type = abm_player_coalition
				target = $first$
			}
		}
		is_rebel_country = no
	}
	can_create_subject_alliance = {
		actor = $first$
		recipient = $second$
	}
	# 1.4 vanilla check (kept as vanilla): no alliances across the Hussite Wars sides.
	trigger_if = {
		limit = {
			situation:hussite_wars ?= { situation_is_active = yes }
		}
		custom_tooltip = {
			text = not_on_opposing_hussite_wars_sides_tt
			NOT = {
				hussite_wars_on_opposing_sides = {
					first = $first$
					second = $second$
				}
			}
		}
	}
	# Ars Belli: vanilla's "no two hegemons" check (NOT = { $first$ and $second$ both is_hegemon })
	# is left out on purpose - hegemons may ally each other (bdfd863).
}

is_mp_gp = {
	has_variable = mp_is_gp
}
is_mp_major = {
	has_variable = mp_is_major
}
is_mp_small = {
	has_variable = mp_is_small
}
is_mp_minor = {
	has_variable = mp_is_minor
}
# Top 3 MP Great Powers by power score (mp_rank_count = countries ranked above us).
# Used by cb_hegemon (casus_belli/hegemon.txt) in place of vanilla is_hegemon.
is_mp_top3_gp = {
	is_mp_gp = yes
	has_variable = mp_rank_count
	var:mp_rank_count < 3
}
is_mp_normal = {
	OR = {
		has_variable = mp_is_normal
		not = { has_variable = mp_is_gp }
		not = { has_variable = mp_is_major }
		not = { has_variable = mp_is_small }
		not = { has_variable = mp_is_minor }
	}
}

# Traditional economy (societal value positive side = right-hand term, i.e. traditional)
# opens up more of the countryside for settlement: population capacity, prosperity and
# production efficiency in rural locations only, applied through a location_modifier with
# a potential_trigger. Upkeep efficiency has NO rural-scoped key (only the country-wide
# `building_upkeep_efficiency`), so that half sits in country_modifier and helps towns too.
abm_traditional_land_tenure = {
	icon = agrarian_community
	societal_values = {
		traditional_economy_focus
	}

	potential = {
		societal_value:capital_economy_vs_traditional_economy >= 65
	}

	# Costs no reform slot: the reform hands back the slot it occupies.
	country_modifier = {
		government_reform_slots = 1
	}

	location_modifier = {
		local_population_capacity_modifier = 0.25
		local_monthly_prosperity = 0.001
		local_production_efficiency = medium_production_efficiency_bonus

		potential_trigger = {
			custom_tooltip = {
				text = abm_rural_free_land_tt
				location_rank ?= location_rank:rural_settlement
			}
		}
	}

	years = 2
}

# Urban counterpart of abm_traditional_land_tenure, on the Capital Economy side (<= -65). `is_city = yes` is the vanilla urban test (towns, cities and megalopolises).
# Growth, not capacity: pop capacity only bites at the ceiling, which towns are far from,
# and there is no non-rural growth key - hence local_population_growth on an is_city
# location_modifier. +0.001 a year is ~+40% relative on a typical ~0.25%/yr location.
abm_guild_charters = {
	icon = urban_guilds
	societal_values = {
		capital_economy_focus #ABM was traditional_economy_focus; the reform needs 65+ toward Capital Economy
	}

	potential = {
		societal_value:capital_economy_vs_traditional_economy <= -65
	}

	# Costs no reform slot: the reform hands back the slot it occupies.
	country_modifier = {
		government_reform_slots = 1
		global_non_rural_monthly_prosperity = 0.001
		raw_material_in_province_impact = 0.025
	}

	location_modifier = {
		local_population_growth = 0.001
		potential_trigger = {
			custom_tooltip = {
				text = abm_urban_growth_tt
				is_city = yes
			}
		}
	}

	years = 2
}

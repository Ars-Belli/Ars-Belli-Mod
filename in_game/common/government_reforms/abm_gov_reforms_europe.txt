REPLACE:merchant_republic = {
	major = yes
	government = republic

	potential = {
		OR = {
			# Christian nations of western or eastern Europe: no societal value requirement.
			AND = {
				religion.group = religion_group:christian
				original_capital ?= {
					OR = {
						sub_continent = sub_continent:western_europe
						sub_continent = sub_continent:eastern_europe
					}
				}
			}
			# North Africa and the Middle East: any religion, but a thoroughly mercantile society.
			# (Societal values run -100..100, so "50+" is the strong half of each axis.)
			AND = {
				original_capital ?= {
					OR = {
						sub_continent = sub_continent:north_africa
						sub_continent = sub_continent:middle_east
					}
				}
				societal_value:serfdom_vs_free_subjects >= 50
				societal_value:aristocracy_vs_plutocracy >= 50
				societal_value:individualism_vs_communalism >= 50
				societal_value:capital_economy_vs_traditional_economy >= 50
			}
		}
	}

	country_modifier = {
		global_merchant_capacity_modifier = 0.50
		trade_income = 0.20
		global_foreign_build_buildings_efficiency = 1.0

		merchant_maintenance_efficiency = 0.50
		#global_integration_speed_modifier = -0.25
		court_language_is_market_language_importance_modifier = 15

	}

	years = 2
}

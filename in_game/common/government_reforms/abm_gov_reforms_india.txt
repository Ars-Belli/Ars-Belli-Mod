abm_bengaleese_meritocracy = {
	major = yes
	unique = yes
	age = age_1_traditions
	potential = {
		OR	=	{
			culture = culture:orissan
			culture = culture:bengali
			culture = culture:zo_culture
			culture = culture:tripuri
			culture = culture:burmese_culture
			has_or_had_tag = BNG
		}
		num_of_ports > 2
		OR = {
			government_type = government_type:monarchy
			government_type = government_type:republic
		}
	}
	allow = {
		societal_value:naval_vs_land > 50
	}
	locked = {
		always = yes
	}
	country_modifier = {
		monthly_towards_naval = societal_value_large_monthly_move
		navy_galley_power = 0.15
		own_coast_naval_combat_bonus = 0.15
	}

	on_activate = {
		hidden_effect = {
			if = {
				limit = { exists = succession_law }
				set_variable = { name = previous_heir_selection value = succession_law }
			}
		}
		change_heir_selection = heir_selection:admiralty_regime_heir_selection
	}
	on_deactivate = {
		if = {
			limit = { has_variable = previous_heir_selection }
			change_heir_selection = var:previous_heir_selection
			remove_variable = previous_heir_selection
		}
		else = {
			set_to_default_heir_selection = yes
		}
	}
	
	years = 2
}

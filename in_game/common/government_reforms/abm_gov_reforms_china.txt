# Ars Belli: vanilla warring_state with the dead red_turban_rebellions branches removed.
# The situation is emptied by this mod, so both branches were always false while
# is_situation_active logged an error every evaluation.
REPLACE:warring_state = {
	age = age_1_traditions
	potential = {
		NOT = { has_or_had_tag = CHI }
		has_reform = government_reform:warring_state
		has_variable = unlocked_warring_state
		trigger_if = {
			limit = { exists = international_organization:middle_kingdom }
			NOT = { is_leader_of_international_organization = international_organization:middle_kingdom }
		}
		OR = {
			culture.language.language_family ?= language_family:chinese_language_family
			culture.language = language:yi_language
			culture.language = language:hmong_language
			culture.language.language_family ?= language_family:tai_language_family
			culture.language = language:tibetan_language
			culture.language = language:jurchen_language
			culture.language = language:nanai_language
			culture.language = language:udege_language
		}
	}
	locked = {
		NOT = { has_or_had_tag = CHI }
		has_reform = government_reform:warring_state
		exists = international_organization:middle_kingdom
		OR = {
			culture.language.language_family ?= language_family:chinese_language_family
			culture.language = language:yi_language
			culture.language = language:hmong_language
			culture.language.language_family ?= language_family:tai_language_family

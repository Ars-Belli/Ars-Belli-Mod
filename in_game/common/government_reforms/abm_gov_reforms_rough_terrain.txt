# Ars Belli - rough-terrain homeland reforms.
#
# Every advance that used to grant a country-wide <topography>_proximity_impact
# bonus now unlocks one of the reforms below instead (see in_game/common/advances/).
# The bonus is applied through location modifiers, so the country only gets it in
# the locations the reform selects, never country-wide.
#
# - abm_homeland_proximity_reform: the region of the country's original capital.
# - abm_alpine_proximity_reform: the same set, unlocked by Alpine advances for
#   Savoy, the Swiss and Provence.
#
# mountains_proximity_impact / plateau_proximity_impact / hills_proximity_impact
# only exist at country scope (main_menu/common/modifier_type_definitions), so
# each rough terrain gets its own location_modifier block instead, holding the
# exact inverse of that terrain's own penalties from in_game/common/topography:
#   proximity:        local_distance_from_capital_speed_propagation = 1/speed - 1
#                     (mountains 0.50 -> +100%, hills 0.75 -> +33%, plateau 0.875 -> +14%)
#   development:      local_monthly_development_modifier = -penalty
#                     (mountains -0.50, hills -0.25, plateau -0.10)
#
# Shapes copied from vanilla:
# - location_modifier + potential_trigger + custom_tooltip:
#   in_game/common/laws/10_italian_wars_laws.txt (italian_catepanate_policy)
# - several location_modifier blocks on one object, with location-scope triggers
#   (vegetation/climate) inside potential_trigger:
#   in_game/common/estate_privileges/tribes_estate.txt (desert_power)
# Both reforms also grant government_reform_slots = 1 (vanilla peasant_republic_reform,
# constitutional_parliament...), so they are slot-neutral: taking one never costs a slot.
# Shared location conditions: in_game/common/scripted_triggers/abm_rough_terrain_triggers.txt
abm_homeland_proximity_reform = {
	unique = yes
	age = age_1_traditions

	potential = {
		OR = {
			has_advance = arm_mount_ararat
			has_advance = tre_pontic_trail_mastery
			has_advance = ira_persian_plateau
			has_advance = abm_che_wetland_adaptation
			has_advance = abm_hng_mountain_adaptation
			has_advance = hab_expert_mountaineers
			has_advance = albanian_prokletije
			has_advance = scottish_highland_tradition
			has_advance = drokpa
			has_advance = inc_vertical_economy
			has_advance = mor_atlas_strongholds
			has_advance = a_permanent_capital
			has_advance = npl_himalayan_homeland
		}
	}

	location_modifier = {
		potential_trigger = {
			abm_rough_terrain_mountains_in_original_capital_region = yes
		}
		# Each value cancels the terrain's own penalty exactly: the proximity term is
		# 1 / terrain_speed - 1 (mountains 0.50, hills 0.75, plateau 0.875), the
		# development term is -1 x the terrain's local_monthly_development_modifier.
		local_distance_from_capital_speed_propagation = 1.0
		local_monthly_development_modifier = 0.5
	}

	location_modifier = {
		potential_trigger = {
			abm_rough_terrain_hills_in_original_capital_region = yes
		}
		local_distance_from_capital_speed_propagation = 0.333
		local_monthly_development_modifier = 0.25
	}

	location_modifier = {
		potential_trigger = {
			abm_rough_terrain_plateaus_in_original_capital_region = yes
		}
		local_distance_from_capital_speed_propagation = 0.143
		local_monthly_development_modifier = 0.10
	}

	country_modifier = {
		# Pays for its own reform slot, so taking it never costs a slot.
		government_reform_slots = 1
		pure_tooltip_entry = abm_homeland_proximity_reform_tt
	}

	years = 2
	ai_will_do = { add = 10 }
}

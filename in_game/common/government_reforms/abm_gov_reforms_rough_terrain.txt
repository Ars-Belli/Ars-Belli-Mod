# Ars Belli - rough-terrain homeland reform.
#
# Every advance that used to grant a country-wide <topography>_proximity_impact
# bonus now unlocks this single reform instead (see in_game/common/advances/). The
# bonus is a location modifier: the country only gets the proximity benefit inside
# the region of its own home capital - the historical capital if it has one,
# otherwise the original capital.
#
# mountains_proximity_impact / plateau_proximity_impact / hills_proximity_impact
# only exist at country scope (main_menu/common/modifier_type_definitions), so the
# local equivalent is local_distance_from_capital_speed_propagation ("Local
# Proximity Efficiency", location scope). +100% cancels the rough-terrain proximity
# cost of the location, and does nothing on flat land.
#
# Shapes copied from vanilla:
# - location_modifier + potential_trigger + custom_tooltip:
#   in_game/common/laws/10_italian_wars_laws.txt (italian_catepanate_policy)
# - save_temporary_scope_as inside a trigger:
#   in_game/common/scripted_triggers/00_event_illustration_triggers.txt
# - comparing a region against a scope's region:
#   in_game/common/country_interactions/invite_prince.txt (capital ?= { region = scope:actor.capital.region })
# - is_historical_capital at location scope:
#   in_game/common/building_types/capital_buildings.txt (location_potential)

abm_homeland_proximity_reform = {
	unique = yes
	age = age_1_traditions

	potential = {
		OR = {
			has_advance = arm_mount_ararat
			has_advance = tre_pontic_trail_mastery
			has_advance = ira_persian_plateau
			has_advance = abm_che_wetland_adaptation
			has_advance = abm_hng_mountain_adaptation
			has_advance = hab_expert_mountaineers
			has_advance = swiss_alpine_passes
			has_advance = sav_alpine_identity
			has_advance = albanian_prokletije
			has_advance = scottish_highland_tradition
			has_advance = drokpa
			has_advance = inc_vertical_economy
			has_advance = mor_atlas_strongholds
			has_advance = a_permanent_capital
			has_advance = npl_himalayan_homeland
		}
	}

	location_modifier = {
		potential_trigger = {
			save_temporary_scope_as = abm_homeland_location
			# The region of the historical capital wins when the country has one.
			trigger_if = {
				limit = {
					owner ?= {
						any_owned_location = {
							is_historical_capital = yes
						}
					}
				}
				custom_tooltip = {
					text = abm_tt_homeland_historical_capital_region
					owner ?= {
						any_owned_location = {
							limit = {
								is_historical_capital = yes
								region = scope:abm_homeland_location.region
							}
						}
					}
				}
			}
			# Otherwise we fall back to the region of the original capital.
			trigger_else = {
				custom_tooltip = {
					text = abm_tt_homeland_original_capital_region
					owner ?= {
						original_capital ?= {
							region = scope:abm_homeland_location.region
						}
					}
				}
			}
		}
		# mountains_/plateau_/hills_proximity_impact are country-only modifiers
		# (modifier_type_definitions: category = country), so they are ignored in a
		# location_modifier. The location-scope equivalent of "rough terrain costs no
		# proximity here" is this: +100% cancels the mountains penalty (-50%) and
		# clamps for hills (-25%) and plateau (-12.5%); flatland has no penalty to
		# cancel, so it gains nothing.
		local_distance_from_capital_speed_propagation = 1.0
	}

	country_modifier = {
		pure_tooltip_entry = abm_homeland_proximity_reform_tt
	}

	years = 2
	ai_will_do = { add = 10 }
}

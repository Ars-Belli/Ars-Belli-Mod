# Settle the Steppe: a cabinet action that pulls settlers into the empty
# grassland of the Pontic steppe.
#
# Deliberately modelled on vanilla's swe_slash_and_burn_new_farms
# (in_game/common/cabinet_actions/): a province picker, per-location growth and
# migration modifiers while the action runs, and an is_finished condition that
# ends it once the province is settled. Settlement is measured in DEVELOPMENT
# only (every land location >= 20.01); the pop half stops by itself once a
# location holds 30 (thousand) pops - a fixed floor, not a share of capacity -
# while development keeps going.
#
# Differences from the vanilla reference:
#   - gated on region:steppes_region instead of a hardcoded area list, so it
#     works for every steppe province without naming them one by one;
#   - the target province must be culturally OURS already: every owned land
#     location in it must have a primary or accepted culture as its dominant
#     culture. Vanilla's action colonises culturally foreign land, which is not
#     what we want here - this is about filling up land we have already
#     assimilated;
#   - and it must be ours in full (abm_owns_entire_province, wasteland aside),
#     matching the full-ownership rule the steppe RGO decisions use.
#
# Unlock: vanilla cabinet actions are unlocked either by a `modifier:allow_...`
# country modifier on an advance, or by `unlock_cabinet_action`. This one uses
# an advance (abm_settle_the_steppe_advance) so it can be researched.
abm_settle_the_steppe = {
	ability = adm
	icon = pop_promote_action
	allow_multiple = yes

	# Any settled government of a non-pagan faith. Same government and religion
	# test the Pontic-steppe decisions use (abm_location_settled_and_loyal), but
	# checked on the country rather than on each location.
	potential = {
		has_advance = abm_settle_the_steppe_advance
		OR = {
			government_type = government_type:monarchy
			government_type = government_type:theocracy
			government_type = government_type:republic
		}
		is_country_religion_pagan = no
	}

	allow = {
		any_province = {
			region = region:steppes_region
			# the whole province, not just the settled part of it
			abm_owns_entire_province = { COUNTRY = root } # allow has no scope:actor
			abm_steppe_province_is_our_culture = yes
			abm_steppe_province_needs_settling = yes
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "abm_settle_the_steppe_select_province"
		none_available_msg_key = "abm_settle_the_steppe_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		column = {
			data = development
		}
		default_sort = development
		visible = {
			# the steppe itself
			any_location_in_province = {
				region = region:steppes_region
			}
			# only land we hold in full, wasteland aside
			abm_owns_entire_province = { COUNTRY = scope:actor }
			# room left to fill: some land location is still below 20 development
			abm_steppe_province_needs_settling = yes
			# and it must already be ours in culture, not a colony project
			abm_steppe_province_is_our_culture = yes

			NOT = {
				cabinet_already_performing_same_task_on_target = {
					cabinet_action = abm_settle_the_steppe
					interaction_target = target
				}
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = province.province_average_development
			}
		}
	}

	# Finished once EVERY land location in the province has reached 20 development.
	# Population plays no part in finishing the action.
	is_finished = {
		scope:target ?= {
			NOT = { abm_steppe_province_needs_settling = yes }
		}
	}

	# Development always runs, so the province reaches the 20-development finish
	# even if the population part below has switched itself off.
	# 0.035/month = one point per ~28.5 months, i.e. ~30 years for a province whose
	# settlements start at 6-9 development (14 points / 0.035 = 400 months, 11.5 / 0.035
	# = 329). The province ends when its LAST location reaches 20.01, so the slowest
	# location sets the duration. Raise/lower this one number for a different pace.
	location_modifier = {
		local_monthly_development = 0.035
	}

	# Pops only while the location is still small: a FIXED floor of 30 (thousand) pops,
	# deliberately not a share of population capacity - capacity grows with development
	# and with buildings, so a percentage would keep re-arming itself and turn this into a
	# cheap pop pump after every epidemic, war or resettlement. Above the floor the growth
	# and migration attraction stop while development keeps going.
	# `potential_trigger` is the supported gate for a cabinet action's location modifier
	# (common/cabinet_actions/readme.txt).
	location_modifier = {
		potential_trigger = {
			population < 30
		}
		local_population_growth = 0.020
		local_migration_attraction = 0.250
	}

	map_marker = {
		show_for_owner = yes
	}
}

# Ars Belli — reform_taxation_system cabinet action disabled.
# Vanilla gated this action on the "Crisis of the Chinese Dynasty" disaster
# (any_active_disaster = disaster_type:crisis_of_the_chinese_dynasty). That
# disaster is removed in Ars Belli (disasters/crisis_of_the_chinese_dynasty.txt
# is an empty placeholder), which left a dangling type reference in the vanilla
# potential. Disabled here rather than re-enabled globally.

reform_taxation_system = {
	ability = adm
	icon = taluqdar_tax_collection
	potential = {
		always = no
	}

	allow = {
		is_during_bankruptcy = no
	}
	days = 1825
	allow_multiple = no
	country_modifier = {
		monthly_celestial_authority = 0.05
	}

	on_activate = {
		scope:actor = {
			add_country_modifier = {
				modifier = chi_disaster_reform_taxation_modifier
				years = 5
				mode = add_and_extend
			}
		}
	}
	on_fully_activated = {
		scope:actor = {
			trigger_event_non_silently = crisis_of_the_chinese_dynasty.7
		}
	}
	on_deactivate = {
		if = {
			limit = { has_country_modifier = chi_disaster_reform_taxation_modifier }
			remove_country_modifier = chi_disaster_reform_taxation_modifier
		}
	}
}

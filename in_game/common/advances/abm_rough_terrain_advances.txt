# Ars Belli - rough-terrain advances (vanilla overrides).
#
# These vanilla advances used to grant country-wide <topography>_proximity_impact
# bonuses. The bonus is replaced by unlock_government_reform: the advance now
# unlocks a homeland reform whose benefit is limited to the historical home region
# of the country (location_modifier + potential_trigger in
# in_game/common/government_reforms/abm_gov_reforms_rough_terrain.txt).
#
# Each block is a REPLACE: of the vanilla entry, copied verbatim minus the
# proximity modifiers - keep it in sync with vanilla when the game updates.

REPLACE:ira_persian_plateau = {
	age = age_1_traditions
	potential = {
		OR = {
			has_or_had_tag = IRA
			culture = { has_culture_group = culture_group:iranian_group }
		}
	}
	unlock_government_reform = abm_homeland_proximity_reform

	requires = ira_shahnameh
}

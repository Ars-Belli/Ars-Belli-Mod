# Ars Belli - rough-terrain advances (vanilla overrides).
#
# These vanilla advances used to grant country-wide <topography>_proximity_impact
# bonuses. The bonus is replaced by unlock_estate_privilege: the advance now unlocks
# a peasants privilege whose benefit is limited to the original capital's region
# of the country (location_modifier + potential_trigger in
# in_game/common/estate_privileges/abm_rough_terrain_privileges.txt).
#
# Each block is a REPLACE: of the vanilla entry, copied verbatim minus the
# proximity modifiers - keep it in sync with vanilla when the game updates.
#
# Exception: swiss_alpine_passes and sav_alpine_identity unlock the alpine
# privilege (abm_alpine_proximity_privilege) instead; it is also limited to the
# original capital's region (Provence joins them through abm_tarascon_castle).
#
# Every advance in this file is age_1_traditions and requires an age_1 advance, so the
# privilege is available from the start (same for the mod-owned unlockers elsewhere).

REPLACE:ira_persian_plateau = {
	age = age_1_traditions
	icon = feudalism_advance
	potential = {
		OR = {
			has_or_had_tag = IRA
			culture = { has_culture_group = culture_group:iranian_group }
		}
	}
	unlock_estate_privilege = abm_homeland_proximity_privilege

	requires = cultural_traditions_law_advance
}

# Swiss - Alpine-advance variant, limited to the original capital's region
# (estate_privileges/abm_rough_terrain_privileges.txt).
# Ars Belli: no REPLACE: - the mod's same-name culture_high_alemannic.txt hides the game's definition, so this is the only one.
swiss_alpine_passes = {
	age = age_1_traditions
	icon = feudalism_advance
	potential = {
		original_capital ?= {
			OR = {
				area = area:western_switzerland_area
				area = area:central_switzerland_area
				area = area:eastern_switzerland_area
			}
		}
		is_swiss_for_advances = yes
	}
	unlock_estate_privilege = abm_alpine_proximity_privilege

	requires = mapmaking
}

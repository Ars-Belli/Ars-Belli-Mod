# Ars Belli urbanisation limit - the research term. +5 Urban Capacity rides on one root
# advance (depth = 0) of each age instead of on a standalone advance, so it comes with
# a technology the country takes anyway:
#
#   age 1  agriculture_advance        Agriculture          starting_technology_level = 1, so most countries start with it
#   age 2  renaissance_development    Scholarly Treatises  no institution needed
#   age 3  printing_press_advance     Printing Press       needs the Printing Press institution embraced
#   age 4  global_trade_advance       Global Trade         needs the Global Trade institution embraced
#   age 5  manufactories_advance      Manufactories        needs the Manufactories institution embraced
#   age 6  industrialization_advance  Industrialization    needs the Industrialization institution embraced
#
# INJECT: adds the one line to each vanilla definition (advances\0_age_of_*.txt) and leaves
# the rest of the advance to vanilla, so a game update needs no re-sync here.
#
# Base 10 + 6 x 5 = 40 town points from base and research at full tech, plus 1 per 20
# locations, 1 per 400k people, and +1 / +2 / +4 for duchy / kingdom / empire rank.

@abm_urbanisation_limit_increase = 5

INJECT:agriculture_advance = {
	abm_urbanisation_limit = @abm_urbanisation_limit_increase
}

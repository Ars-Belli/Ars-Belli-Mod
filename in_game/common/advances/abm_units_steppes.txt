# Advances that unlock the steppe cultural regiments in
# in_game/common/unit_types/abm_units_steppes.txt.
#
# One advance per regiment, so a unit never appears in the build list before
# its own age's upgrade of the same role has been researched (every
# `requires` below points at the vanilla advance that unlocks the baseline
# unit the regiment copies from, and sits in the same age).
#
# The `potential` mirrors each unit's own `country_potential` - culture, or
# accepted culture with a positive cultural view, or the Cossacks estate for
# the Cossack lines - and then adds the steppe capital requirement, so moving
# the seat into the steppe later does not retroactively unlock them.

#############################
# Crimean
#############################

abm_crimean_horse_archers_2_advance = {
	content_priority = 900
	age = age_2_renaissance
	icon = glorious_arms

	potential = {
		OR = {
			culture = culture:crimean
			AND = {
				has_accepted_culture = culture:crimean
				culture = {
					cultural_view = {
						target = culture:crimean
						value >= positive
					}
				}
			}
		}
		original_capital ?= {
			region = region:steppes_region
		}
	}

	requires = unlock_cavalry_advance

	unlock_unit = abm_crimean_horse_archers_2
}

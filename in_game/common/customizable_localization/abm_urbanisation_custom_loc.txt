# Ars Belli urbanisation limit - the top-right readout.
#
# Same shape as mp_limits_custom_loc.txt: one entry that picks a coloured
# localization key from a trigger, so the counter in panels\right_panel\right_panel.gui
# turns yellow at the cap and red past it without any GUI-side branching.
#
# First matching text wins, so the order here is over -> at -> under.

abm_urbanisation_display = {
	type = country
	text = {
		localization_key = ABM_URBAN_CAPACITY_OVER
		trigger = {
			abm_urbanisation_free_points < 0
		}
	}
	text = {
		localization_key = ABM_URBAN_CAPACITY_FULL
		trigger = {
			abm_urbanisation_free_points < 1
		}
	}
	text = {
		localization_key = ABM_URBAN_CAPACITY_NORMAL
	}
}

# Label for the tier list panel's Urban Capacity popup toggle - same shape as
# mp_fort_alert_toggle_label in mp_limits_custom_loc.txt.
abm_urban_alert_toggle_label = {
	type = country
	text = {
		localization_key = ABM_URBAN_ALERT_TOGGLE_OFF
		trigger = { has_variable = abm_urban_alert_muted }
	}
	text = {
		localization_key = ABM_URBAN_ALERT_TOGGLE_ON
	}
}

# ARS BELLI: scripted replacement for the engine "Send Economic Support".
#
# The engine version is untouchable from script - it is not a relation_type, has no
# scripted_relation, country_interaction, on_action, ai_diplochance entry or allowed_*
# modifier, and its action list is enumerated by the engine - so none of the three
# requirements below could be attached to it. This rebuilds it as a scripted relation,
# which gets all three for free. The vanilla action is marked as against the rules in
# localisation instead, since it cannot be removed.
#
#   Nr.3  the offer flow gives the recipient a native accept/decline popup
#   Nr.4  giving OR receiving costs a Defensive Point (see the mp weight recompute in
#         scripted_effects/ars_belli_scripted_effects.txt, which spills into Alliance
#         Points via the existing overflow rules)
#   Nr.5  -0.10 monthly diplomats on BOTH sides via the giving_/receiving_ auto-modifiers
#         in static_modifiers/ars_belli_country_changes.txt
#
# first = the country sending gold, second = the country receiving it.

ars_belli_economic_support = {
	type = diplomacy
	relation_type = oneway

	category = {
		offer = CATEGORY_ECONOMY_ACTIONS
		request = CATEGORY_ECONOMY_ACTIONS
		cancellation = CATEGORY_ECONOMY_ACTIONS
		break = CATEGORY_ECONOMY_ACTIONS
	}

	break_on_war = yes
	annulled_by_peace_treaty = yes

	# The support itself. Vanilla lets the sender dial an amount; a scripted relation has
	# nowhere to persist a per-pair figure, so this scales off the sender's income instead.
	gold_to_second = ars_belli_economic_support_amount

	visible = {
		scope:actor = {
			has_limited_diplomacy = no
		}
	}

	# Sender-initiated only, matching the vanilla action it replaces.
	request_visible = { always = no }

	offer_enabled = {
		custom_tooltip = {
			text = ars_belli_economic_support_points_tt
			scope:actor = {
				NOT = { is_over_defensive_limit = 1 }
			}
		}
		custom_tooltip = {
			text = ars_belli_economic_support_recipient_points_tt
			scope:recipient = {
				NOT = { is_over_defensive_limit = 1 }
			}
		}
	}

	# AI accepts economic support as a matter of course - it is free gold - and only
	# refuses when taking it would push it past its own limits.
	wants_to_receive = {
		add = {
			desc = "BASE_VALUE"
			value = 1000
		}
		if = {
			limit = {
				scope:actor = { is_over_defensive_limit = 1 }
			}
			subtract = {
				desc = ars_belli_economic_support_points_tt
				value = 5000
			}
		}
		if = {
			limit = {
				scope:actor = { is_over_alliance_limit = 1 }
			}
			subtract = {
				desc = ars_belli_economic_support_points_tt
				value = 5000
			}
		}
	}

	wants_to_receive_diplo_chance = {
		base = 1000
	}

	# Keep it unless it starts costing us points we cannot afford.
	wants_to_keep = {
		add = {
			desc = "BASE_VALUE"
			value = 100
		}
		if = {
			limit = {
				scope:actor = { is_over_alliance_limit = 1 }
			}
			subtract = {
				desc = ars_belli_economic_support_points_tt
				value = 500
			}
		}
	}
}

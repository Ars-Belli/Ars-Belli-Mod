# or_qapi_province (kalay kezlev aqmecit jurchi or_qapi) is in crimea_area,
# steppes_region. The decision is gated by abm_pontic_steppe.txt.
#
# kezlev is the only location this changes; kalay, aqmecit, jurchi and
# or_qapi keep their vanilla goods.
#
# Campaign-global, not per country: only_once stops the same country from
# taking it twice, and abm_or_qapi_rgos_developed is the global flag that
# removes the decision for every other country as well - including any
# country that later conquers the province or has its RGOs changed again.
abm_develop_or_qapi_rgos = {
	decision_category = economy_decisions
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_stone.dds"
	only_once = yes

	potential = {
		NOT = { has_global_variable = abm_or_qapi_rgos_developed }

		owns = location:kezlev

		NOT = { location:kezlev = { raw_material = goods:stone } }
	}

	allow = {
		abm_can_develop_or_qapi_rgos = yes
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = abm_develop_or_qapi_rgos.a
		ai_chance = { value = 100 }

		effect = {
			set_global_variable = abm_or_qapi_rgos_developed

			location:kezlev = {
				change_max_raw_material_workers = {
					value = rgo_workers
					subtract = 1
					min = 0
					multiply = -1
				}
				change_raw_material = goods:stone
			}
		}
	}
}

# Entry point for the steppe pastoral RGO reorganisation. The decision only
# prompts: its option fires the generic action, which walks the player through
# province -> location -> goods. `owncountry` generic actions have no automatic
# UI, so this decision is what makes the feature discoverable.
# Repeatable on purpose - the action has no cooldown, and `potential` already
# disappears once no province qualifies.
abm_reorganize_pastoral_rgo_decision = {
	decision_category = economy_decisions
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_wool.dds"

	potential = {
		any_province = {
			abm_steppe_pastoral_province = { COUNTRY = root }
		}
	}

	allow = {
		any_province = {
			abm_steppe_pastoral_province = { COUNTRY = root }
		}
	}

	ai_will_do = {
		value = 0
	}

	option = {
		name = abm_reorganize_pastoral_rgo_decision.a
		ai_chance = { value = 0 }

		effect = {
			fire_generic_action = {
				type = abm_reorganize_pastoral_rgo
				actor = root
			}
		}
	}
}

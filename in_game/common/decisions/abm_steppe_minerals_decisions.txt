# Historical mineral decisions for the Pontic-Caspian steppes.
#
# These convert pastoral / farming locations that sat on documented mineral
# deposits: the Donets coalfields (Luhansk, Bakhmut, Kalmius), Donets Ridge and
# Kursk Magnetic Anomaly iron, the northern forest-steppe timber, and the
# non-ferrous metals of the Caucasus foothills (copper, lead) and the
# Orenburg / Ural margin (copper, tin).
#
# Same shape as the three Crimea decisions in abm_pontic_steppe_decisions.txt:
# free, campaign-global once-only via a global flag, and gated by
# abm_can_develop_province_rgos (parameterised province variant of
# abm_can_develop_*_rgos) so every owned land location in the province must be
# a full core of a settled, non-pagan, culturally and religiously loyal owner.

# Luhansk coalfield: the western end of the Donets basin, worked from the
# 1790s Lugansk foundry onwards.
# Campaign-global once-only, free, like the three Crimea decisions.
abm_develop_luhan_rgos = {
	decision_category = economy_decisions
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_coal.dds"
	only_once = yes

	potential = {
		NOT = { has_global_variable = abm_luhan_rgos_developed }

		owns = location:luhansk
		owns = location:mityakinskaya

		OR = {
			NOT = { location:luhansk = { raw_material = goods:coal } }
			NOT = { location:mityakinskaya = { raw_material = goods:coal } }
		}
	}

	allow = {
		abm_can_develop_province_rgos = { PROVINCE = province:luhan_province }
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = abm_develop_luhan_rgos.a
		ai_chance = { value = 100 }

		effect = {
			set_global_variable = abm_luhan_rgos_developed

			location:luhansk = {
				change_max_raw_material_workers = {
					value = rgo_workers
					subtract = 1
					min = 0
					multiply = -1
				}
				change_raw_material = goods:coal
			}
			location:mityakinskaya = {
				change_max_raw_material_workers = {
					value = rgo_workers
					subtract = 1
					min = 0
					multiply = -1
				}
				change_raw_material = goods:coal
			}
		}
	}
}

# ARS BELLI: "Make Available For Hire" is removed.
#
# Renting your own regiments out to another country prints money: the mercenary modifiers
# (upkeep, morale, the lot) land on the hirer, while the owner keeps drawing the full hire
# price for troops it no longer pays for. Between two players that is a straight gold
# transfer with a profit attached, which is exactly what the gift and economic support caps
# elsewhere in the mod exist to prevent.
#
# The gate is in potential rather than allow, so the action drops out of the list entirely
# instead of sitting there permanently greyed out. That covers the player, the delegation
# automation and any AI evaluation, since all three reach the action through this same gate.
# Vanilla already had ai_tick = never here, which per the country_interactions readme marks
# an action "already handled in code" - so if AI countries list mercenaries through engine
# code rather than through this action, that path is not reachable from script and is not
# something this change (or any data change) can close.
#
# delist_unit lives in the same vanilla file but is a separate key, so this REPLACE leaves it
# untouched: anyone who listed a unit before this change still needs a way to pull it back.
#
# The body below is vanilla's, unchanged except for the potential block, so that nothing
# depends on fields being dropped.
REPLACE:make_unit_available_for_hire = {
	type = owncountry

	potential = {
		always = no
	}

	allow = {
	}

	ai_prerequisite = {
		always = no
	}

	ai_tick = never
	ai_tick_frequency = 90

	automation_tick = daily
	automation_tick_frequency = 20

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = target
		name = "make_unit_available_for_hire_select_unit"
		none_available_msg_key = "make_unit_available_for_hire_no_units"
		pre_evaluation_sort_value = {
			#use fixed AI override values (multiplier=2, duration=24) to cheaply rank units
			#before committing to a full ai_will_do evaluation; mirrors the scoring in ai_will_do
			value = "hire_price(2|24)"
		}
		pre_evaluation_number_to_evaluate_fully = 5
		column = {
			data = name
		}
		column = {
			data = unit_location
		}
		visible = {
			always = no
		}
	}

	select_trigger = {
		looking_for_a = value
		target_flag = target_1
		name = "make_unit_available_for_hire_select_cost_multiplier"
		min = 0
		max = 5
		step = 0.05
		default = 1
		ai_override_value = 2 #for the AI, only test this value, we want it to be a bit more performant
	}

	select_trigger = {
		looking_for_a = value
		target_flag = target_2
		name = "make_unit_available_for_hire_select_duration"
		min = 1
		max = "define:NMercenary|MERCENARY_MAX_DURATION_IN_MONTHS"
		step = 1
		default = "define:NMercenary|MERCENARY_DEFAULT_DURATION_IN_MONTHS"
		ai_override_value = "define:NMercenary|MERCENARY_DEFAULT_DURATION_IN_MONTHS" #for the AI, only test this value, smaller durations aren't worth testing and we want it to be a bit more performant
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
			}
			scope:target = {
				make_unit_available_for_hire = {
					cost_multiplier = scope:target_1
					months = scope:target_2
					save_scope_as = merc_for_hire
				}
			}
		}
	}

	ai_will_do = {
		add = -1000
	}
}

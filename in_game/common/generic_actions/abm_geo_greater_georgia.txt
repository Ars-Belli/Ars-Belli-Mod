# Georgia (GEO): claim the inheritance of the united Georgian kingdom.
#
# scope:actor = country (GEO)
#
# ">80% of the Caucasus region as cores" counts owned AND cored locations of
# region:caucasus_region. It holds 241 locations (map_data/definitions.txt),
# so 80% = 192.8 -> 193 locations are needed for strictly more than 80%.
abm_geo_greater_georgia = {
	type = owncountry
	icon = restore_rome_primacy
	show_message = no

	potential = {
		scope:actor = {
			has_or_had_tag = GEO
			NOT = { has_reform = government_reform:abm_geo_greater_georgia_reform }
		}
	}

	allow = {
		scope:actor = {
			in_civil_war = no
			NOT = { has_reform = government_reform:abm_geo_greater_georgia_reform }
			custom_tooltip = {
				text = abm_geo_greater_georgia_tt
				region:caucasus_region = {
					any_location_in_region = {
						owner ?= scope:actor
						integration_level = core
						count >= 193
					}
				}
			}
		}
	}

	cooldown = {
		type = abm_geo_greater_georgia
		years = 5
	}

	automation_tick = never
	automation_tick_frequency = 12

	ai_tick = monthly
	ai_tick_frequency = 12

	effect = {
		scope:actor = {
			unlock_government_reform_effect = { type = abm_geo_greater_georgia_reform }
			add_reform = government_reform:abm_geo_greater_georgia_reform

			if = {
				limit = { country_rank_level < 4 }
				set_country_rank = country_rank:rank_empire
			}

			add_prestige = prestige_severe_bonus
			add_government_power = government_power_severe_bonus
		}
	}

	ai_will_do = { value = 100 }
}

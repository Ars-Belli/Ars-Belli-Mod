# Economy projects
ssg_restore_royal_granaries = {
	icon = create_trade
	type = situation
	show_message = no
	potential = {
		scope:actor = {
			has_variable = ssg_jo_restoration_path
			NOT = { has_variable = ssg_economy_stage_1 }
		}
	}
	allow = {
		scope:actor = {
			custom_tooltip = { text = ssg_no_active_project_tt NOT = { has_variable = ssg_active_project } }
			gold >= ssg_project_cost_standard
		}
	}
	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:ssg_hanyang_jo_restoration = this
			situation_is_active = yes
		}
	}
	effect = {
		scope:actor = {
			add_gold = { value = ssg_project_cost_standard multiply = -1 }
			set_variable = { name = ssg_active_project years = 2 }
			trigger_event_silently = { id = flavor_ssg.10 years = 2 }
		}
	}
	ai_tick = monthly
	ai_tick_frequency = 12
	ai_prerequisite = { NOT = { has_game_rule = historical_ai_choices } }
	ai_will_do = { add = 100 }
}

# Reunification requires the four project capstones and effective control of Korea.
ssg_proclaim_reunited_goryeo = {
	icon = claim_high_kingship
	type = situation
	show_message = no
	potential = {
		scope:actor = {
			has_variable = ssg_jo_restoration_path
			has_variable = ssg_economy_complete
			has_variable = ssg_military_complete
			has_variable = ssg_research_complete
			has_variable = ssg_diplomacy_complete
			NOT = { has_variable = ssg_reunited_goryeo }
		}
	}
	allow = {
		scope:actor = {
			custom_tooltip = {
				text = ssg_reunification_independent_tt
				is_subject = no
			}
			custom_tooltip = {
				text = ssg_no_active_project_tt
				NOT = { has_variable = ssg_active_project }
			}
			custom_tooltip = {
				text = ssg_reunification_no_rival_tt
				NOT = { country_exists = c:KOR }
			}
			custom_tooltip = {
				text = ssg_reunification_control_tt
				region:korea_region = {
					any_location_in_region = {
						owner ?= scope:actor
						percent >= 0.95
					}
				}
			}
		}
	}
	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:ssg_hanyang_jo_restoration = this
			situation_is_active = yes
		}
	}
	effect = { scope:actor = { trigger_event_non_silently = flavor_ssg.50 } }
	ai_tick = monthly
	ai_tick_frequency = 12
	ai_prerequisite = { NOT = { has_game_rule = historical_ai_choices } }
	ai_will_do = { add = 100 }
}

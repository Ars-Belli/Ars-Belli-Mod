# Greater Caucasus: any Caucasus-rooted state (other than Georgia or Armenia) that
# unites the whole region and claims a supra-regional crown.
#
# scope:actor = country
#
# Eligibility: original capital in region:caucasus_region, primary culture in the
# Caucasian, Iranian or Turkic group (alan_culture has no group in vanilla, so it
# is matched by name), and not Georgia (GEO) or Armenia (ARM) - those two have
# their own capstone reforms. Only monarchy / theocracy / republic can hold it.
#
# The 80% requirement counts owned AND cored locations of region:caucasus_region.
# It holds 241 locations (map_data/definitions.txt), so 80% = 192.8 -> 193
# locations are needed for strictly more than 80%.
abm_greater_caucasus = {
	type = owncountry
	icon = restore_rome_primacy
	show_message = no

	potential = {
		scope:actor = {
			NOT = { has_reform = government_reform:abm_greater_caucasus_reform }
			NOT = { has_or_had_tag = GEO }
			NOT = { has_or_had_tag = ARM }
			NOT = { has_or_had_tag = CIL }
			original_capital ?= { region = region:caucasus_region }
			OR = {
				culture = { has_culture_group = culture_group:caucasian_group }
				culture = { has_culture_group = culture_group:iranian_group }
				culture = { has_culture_group = culture_group:turkic_group }
				# alan_culture carries no culture group in vanilla, so match it directly.
				culture = culture:alan_culture
			}
		}
	}

	allow = {
		scope:actor = {
			in_civil_war = no
			NOT = { has_reform = government_reform:abm_greater_caucasus_reform }
			OR = {
				government_type = government_type:monarchy
				government_type = government_type:theocracy
				government_type = government_type:republic
			}
			custom_tooltip = {
				text = abm_greater_caucasus_tt
				region:caucasus_region = {
					any_location_in_region = {
						owner ?= scope:actor
						integration_level = core
						count >= 193
					}
				}
			}
		}
	}

	cooldown = {
		type = abm_greater_caucasus
		years = 5
	}

	automation_tick = never
	automation_tick_frequency = 12

	ai_tick = monthly
	ai_tick_frequency = 12

	effect = {
		scope:actor = {
			unlock_government_reform_effect = { type = abm_greater_caucasus_reform }
			add_reform = government_reform:abm_greater_caucasus_reform

			if = {
				limit = { country_rank_level < 3 }
				set_country_rank = country_rank:rank_kingdom
			}

			add_prestige = prestige_severe_bonus
			add_government_power = government_power_severe_bonus
		}
	}

	ai_will_do = { value = 100 }
}

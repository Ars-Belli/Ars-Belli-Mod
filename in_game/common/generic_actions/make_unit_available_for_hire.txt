# ARS BELLI: "Make Available For Hire" is removed.
#
# Renting your own regiments out to another country prints money: the mercenary modifiers
# (upkeep, morale, the lot) land on the hirer, while the owner keeps drawing the full hire
# price for troops it no longer pays for. Between two players that is a straight gold transfer
# with a profit attached, which is exactly what the gift and economic support caps elsewhere
# in the mod exist to prevent.
#
# ---------------------------------------------------------------------------------------
# THIS IS A SAME-NAME FULL-FILE REPLACEMENT, ON PURPOSE. Do not convert it back to a
# `REPLACE:` in a separately-named file.
#
# The first attempt (17a2076, `ars_belli_no_mercenary_listing.txt`, holding
# `REPLACE:make_unit_available_for_hire`) shipped and did nothing at all - the action stayed
# in the game and units could still be listed - and it logged no error of any kind. `REPLACE:`
# demonstrably does work in some databases (the mod's `REPLACE:order_commandery` is live: the
# engine reports it under the plain key in error.log), so this is a per-database gap rather
# than bad syntax, and generic_actions is one of the databases where it does not take.
# Overriding the whole vanilla file by name is the mechanism this folder already relies on for
# `red_turban_rebellions.txt` and `rise_of_timur.txt`, and it needs no `REPLACE:` support.
# ---------------------------------------------------------------------------------------
#
# The action is closed off at every stage rather than at one, because the first attempt gated
# three of them and still shipped a working action, so no single gate is trusted here:
#
#   show_in_gui_list  keeps it out of the automatically generated action lists, which is where
#                     the button comes from - no .gui file references this action by name, so
#                     there is no hand-placed button to remove
#   potential         drops it out of the list entirely rather than leaving it greyed out, and
#                     covers player, AI and delegation automation, which share the gate
#   allow             blocks it if anything reaches it past potential
#   ai_prerequisite   stops the AI evaluating it
#   automation_tick   stops the delegation automation running it
#   select_trigger    offers no unit to pick
#   effect            does nothing even if every gate above is somehow bypassed - this is the
#                     one that guarantees no unit can end up listed
#
# Vanilla already had `ai_tick = never` here, which per the readme marks an action "already
# handled in code". If AI countries list mercenaries through engine code rather than through
# this action, that path is not reachable from script and no data change can close it.
#
# `delist_unit` below is vanilla, untouched: anyone who listed a unit before this change still
# needs a way to pull it back.
make_unit_available_for_hire = {
	type = owncountry

	# Ars Belli: keeps the button out of every auto-generated action list.
	show_in_gui_list = no

	potential = {
		always = no
	}

	allow = {
		always = no
	}

	ai_prerequisite = {
		always = no
	}

	ai_tick = never
	ai_tick_frequency = 90

	# Ars Belli: was daily - stops the delegation automation reaching it.
	automation_tick = never
	automation_tick_frequency = 20

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = target
		name = "make_unit_available_for_hire_select_unit"
		none_available_msg_key = "make_unit_available_for_hire_no_units"
		pre_evaluation_sort_value = {
			#use fixed AI override values (multiplier=2, duration=24) to cheaply rank units
			#before committing to a full ai_will_do evaluation; mirrors the scoring in ai_will_do
			value = "hire_price(2|24)"
		}
		pre_evaluation_number_to_evaluate_fully = 5
		column = {
			data = name
		}
		column = {
			data = unit_location
		}
		# Ars Belli: no unit is ever offered. Vanilla's condition set is kept below, commented
		# out, so it is obvious what was dropped if this ever has to be restored.
		visible = {
			always = no
		}
		# visible = {
		# 	or = {
		# 		is_army = no
		# 		owner = {
		# 			not = { country_type = army }
		# 		}
		# 	}
		# 	or = {
		# 		is_navy = no
		# 		owner = {
		# 			not = { country_type = navy }
		# 		}
		# 	}
		# 	has_mercenaries = no
		# 	has_levies = no
		# 	# A ruler-led unit can not be rented out (would send the head of state to lead a
		# 	# foreign mercenary company); mirrors NUnitUtilities::CanBecomeMercenary.
		# 	trigger_if = {
		# 		limit = { exists = leader }
		# 		leader = { is_ruler = no }
		# 	}
		# }
	}
	
	select_trigger = {
		looking_for_a = value
		target_flag = target_1
		name = "make_unit_available_for_hire_select_cost_multiplier"
		min = 0
		max = 5
		step = 0.05
		default = 1
		ai_override_value = 2 #for the AI, only test this value, we want it to be a bit more performant
	}

	select_trigger = {
		looking_for_a = value
		target_flag = target_2
		name = "make_unit_available_for_hire_select_duration"
		min = 1
		max = "define:NMercenary|MERCENARY_MAX_DURATION_IN_MONTHS"
		step = 1
		default = "define:NMercenary|MERCENARY_DEFAULT_DURATION_IN_MONTHS"
		ai_override_value = "define:NMercenary|MERCENARY_DEFAULT_DURATION_IN_MONTHS" #for the AI, only test this value, smaller durations aren't worth testing and we want it to be a bit more performant
	}

	# Ars Belli: the listing call is gone. This is the backstop - if any path at all still
	# reaches the action, walking it through to the end now lists nothing. Vanilla's body was:
	#
	#	if = {
	#		limit = { exists = scope:target }
	#		scope:target = {
	#			make_unit_available_for_hire = {
	#				cost_multiplier = scope:target_1
	#				months = scope:target_2
	#				save_scope_as = merc_for_hire
	#			}
	#		}
	#	}
	effect = {
	}

	# Ars Belli: vanilla scored this off hire_price against scope:target, which no longer
	# exists here. A flat refusal, matching the shape used for the other disabled actions.
	ai_will_do = {
		add = -1000
	}
}

delist_unit = {
	type = owncountry

	potential = {
		scope:actor = {
			any_rented_out_mercenary = {
				has_customer = no
				is_carrying_or_loaded_unit = no
			}
		}
	}

	allow = {
	}

	ai_tick = never
	ai_tick_frequency = 90

	ai_prerequisite = {
		at_war = yes
	}

	automation_tick = daily
	automation_tick_frequency = 20

	#area in scope:target
	select_trigger = {
		looking_for_a = mercenary
		source = actor
		target_flag = target
		name = "delist_unit_select_merc"
		none_available_msg_key = "delist_unit_no_mercenaries"
		column = {
			data = name
		}
		visible = {
			has_customer = no
			# destroy_mercenary tears the embark link and drowns loaded armies;
			# the C++ CDelistMercenaryCommand path is guarded by NMercenary::CanDelistMercenary,
			# but this script action calls destroy_mercenary directly and would bypass it.
			is_carrying_or_loaded_unit = no
		}
	}
	
	effect = {
		if = {
			limit = {
				exists = scope:target
			}
			scope:target = {
				custom_tooltip = {
					text = delist_unit_merc_destroyed
					destroy_mercenary = scope:target
				}
			}
		}
	}

	ai_will_do = {
		scope:actor = {
			if = {
				limit = {
					at_war = yes
				}
				add = 100
			}
		}
	}
}

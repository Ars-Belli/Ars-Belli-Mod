# ARS BELLI: flat level cap for the trade_office.
#
# Vanilla's trade_office_max_level scaled the cap off the location (development*0.1 + population*0.05
# + a city/megalopolis bonus, plus the owner's market_building_levels), landing around 10 levels in a
# city. Ars Belli uses a plain per-rank cap instead:
#
#   town / rural settlement = 1    city = 2    megalopolis = 3
#
# Those are TOTALS: base 1 plus the rank bonus (city +1, megalopolis +2, exclusive via else_if).
# No development, population or modifier scaling.
#
# The building points at this mod-owned name (REPLACE:trade_office -> max_levels =
# abm_trade_office_max_level) because REPLACE: is per-database and silent when unsupported, so a
# mod-owned value makes the change certain. Vanilla's trade_office_max_level is left untouched.
#
# fondaco is NOT here: it is a flat `max_levels = 1` (one single fondaco per location, never
# upgradeable) - see building_types/abm_foreign_buildings.txt.

abm_trade_office_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	if = {
		limit = {
			location_rank ?= location_rank:megalopolis
		}
		add = {
			desc = "BUILDING_LEVEL_IS_MEGAPOLIS"
			value = 2
		}
	}
	else_if = {
		limit = {
			location_rank ?= location_rank:city
		}
		add = {
			desc = "BUILDING_LEVEL_IS_CITY"
			value = 1
		}
	}
}

# ARS BELLI: age-scaling level cap for the capital manpower buildings - order_headquarters,
# sergeantry, city_guard and temple_militia (all in building_types/abm_capital_buildings.txt).
#
# One step per age: 3 levels in the Age of Traditions and +3 in every following age
# (3 / 6 / 9 / 12 / 15 / 18). Vanilla's janissary_barracks_size is the proven shape: a script
# value that adds levels per age and is referenced by `max_levels = <value name>`. The value is
# evaluated in the building's location scope and current_age_or_later is global, so it works here.
# The cap only grows, so levels already built are never invalidated.
abm_starting_manpower_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 3
	}
	if = {
		limit = {
			current_age_or_later = { age = age_2_renaissance }
		}
		add = {
			desc = age_2_renaissance
			value = 3
		}
	}
	if = {
		limit = {
			current_age_or_later = { age = age_3_discovery }
		}
		add = {
			desc = age_3_discovery
			value = 3
		}
	}
	if = {
		limit = {
			current_age_or_later = { age = age_4_reformation }
		}
		add = {
			desc = age_4_reformation
			value = 3
		}
	}
	if = {
		limit = {
			current_age_or_later = { age = age_5_absolutism }
		}
		add = {
			desc = age_5_absolutism
			value = 3
		}
	}
	if = {
		limit = {
			current_age_or_later = { age = age_6_revolutions }
		}
		add = {
			desc = age_6_revolutions
			value = 3
		}
	}
}

# ARS BELLI: half-size manpower buildings (building_types/abm_small_manpower_buildings.txt).
# Armory, Training Fields, Barracks, Regimental Camp and Conscription Center get a cap of
# 1 + population x 0.05: vanilla's base level of 1, but twice vanilla's population share
# (manpower_max_level is 1 + population x 0.025). The half-size levels double only the
# population part of the cap, so the cap rises in steps half as big.
# Vanilla manpower_max_level is left as it is: a D009 unique barracks and a Mali event read it.

abm_manpower_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		multiply = 0.05
	}
}

# Soldier employment per level of those buildings, half of vanilla manpower_employment (1.0).
abm_half_manpower_employment = 0.5

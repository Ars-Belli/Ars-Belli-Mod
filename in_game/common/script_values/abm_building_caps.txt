# ARS BELLI: flat level cap for the trade_office.
#
# Vanilla's trade_office_max_level scaled the cap off the location (development*0.1 + population*0.05
# + a city/megalopolis bonus, plus the owner's market_building_levels), landing around 10 levels in a
# city. Ars Belli uses a plain per-rank cap instead:
#
#   town / rural settlement = 1    city = 2    megalopolis = 3
#
# Those are TOTALS: base 1 plus the rank bonus (city +1, megalopolis +2, exclusive via else_if).
# No development, population or modifier scaling.
#
# The building points at this mod-owned name (REPLACE:trade_office -> max_levels =
# abm_trade_office_max_level) because REPLACE: is per-database and silent when unsupported, so a
# mod-owned value makes the change certain. Vanilla's trade_office_max_level is left untouched.
#
# fondaco is NOT here: it is a flat `max_levels = 1` (one single fondaco per location, never
# upgradeable) - see building_types/abm_foreign_buildings.txt.

abm_trade_office_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	if = {
		limit = {
			location_rank ?= location_rank:megalopolis
		}
		add = {
			desc = "BUILDING_LEVEL_IS_MEGAPOLIS"
			value = 2
		}
	}
	else_if = {
		limit = {
			location_rank ?= location_rank:city
		}
		add = {
			desc = "BUILDING_LEVEL_IS_CITY"
			value = 1
		}
	}
}

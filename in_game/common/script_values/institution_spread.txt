# ARS BELLI - currently the time it takes for institutions to spread without the 'could have spawned institution' mechanic turned on 
# (which players keep asking for) is too long for certain areas to be viable in a multiplayer setting. As a compromise between said
# mechanic and the old (very high) values, increased trade spread will be tested. As an addition it was made harder to lock out 
# institution spread for enemies by tweaking 'neighbor' spread values and shifting from 'friendly negihbor' to simply 'neighbor'.

#########################################
# Early factors
#########################################
institution_trade_spread_value_early = {
	value = development
	multiply = 0.040 #was 0.03 - 1.33x
	add = {
		value = average_location_literacy
		multiply = 0.100
	}

}

institution_base_spread_from_friendly_neighbor_with_early = {
	value = average_location_literacy
	multiply = 0.030 #was 0.040
}

institution_base_spread_from_neighbor_with_early = {
	value = average_location_literacy
	multiply = 0.015 #was 0.005

}


institution_spread_from_was_possible_spawn_early = {
	value = average_location_literacy
	if = {
		limit = {
			has_game_rule = institution_spawn_random_city
		}
		multiply = 0
	}
	if = {
		limit = {
			has_game_rule = institution_could_have_spawned_spread_disabled
		}
		multiply = 0
	}
	multiply = 0.2
}


institution_total_embraced_to_capital_early = {
	value = average_location_literacy
	multiply = 0.05

}

institution_spread_to_market_member_early = {
	value = average_location_literacy
	multiply = 0.03

}

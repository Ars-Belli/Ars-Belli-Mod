# Ars Belli urbanisation limit - country scope.
#
# Town points: town = 1, city = 2, megalopolis = 4. Both sides of the sum are
# country modifiers, so this is a cheap read and never iterates locations:
#   abm_urbanisation_used  is summed by the engine from the location_ranks
#                          country_modifier blocks (in_game\common\location_ranks).
#   abm_urbanisation_limit is base + locations + advances (auto_modifiers\abm_country.txt).

# How many more town points we can take on. The limit is rounded to a whole number
# first, so the gates, the counter colour and the over-limit penalty all see exactly
# the "X / Y" the counter shows - the size terms are fractional (locations / 20,
# pop / 250), and unrounded, 29 used against 28.7 read as "29 / 29" yet counted as over.
# Can go negative after conquest: that blocks founding and upgrading and costs tax
# efficiency (abm_over_urbanisation_limit in auto_modifiers\abm_country.txt) until we
# are back under (or downgrade something, see generic_actions\abm_downgrade_location_rank.txt).
abm_urbanisation_free_points = {
	value = modifier:abm_urbanisation_limit
	round = yes
	subtract = modifier:abm_urbanisation_used
}

# Display wrappers. The GUI and localization read script values by name through
# [GetPlayer.MakeScope.ScriptValue('...')], which is how the mod already draws the
# multiplayer point counters in panels\right_panel\right_panel.gui, so each of the
# two modifiers gets a named value to be read back through.
abm_urbanisation_used_points = {
	value = modifier:abm_urbanisation_used
}

abm_urbanisation_limit_points = {
	value = modifier:abm_urbanisation_limit
	round = yes
}

# Whole town points held beyond the limit, 0 when at or under it. Scales the
# over-limit penalty, abm_over_urbanisation_limit in auto_modifiers\abm_country.txt.
abm_urbanisation_points_over = {
	value = abm_urbanisation_free_points
	multiply = -1
	min = 0
}

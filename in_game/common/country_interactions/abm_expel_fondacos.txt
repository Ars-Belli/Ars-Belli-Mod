# ARS BELLI: the enforcing half of the fondaco rights system - the fondaco twin of
# abm_expel_trade_offices (read that file for the full rationale). Vanilla allows granting
# `fondaco_rights` but gives no way to clear out a fondaco that has ALREADY been built - revoking
# the rights only stops new ones - so this is that missing lever.
#
# Destroys every fondaco AND hanseatic kontor the target country holds in the ACTOR's own locations
# (they share one slot per location) and revokes their
# vanilla `fondaco_rights` in the same action, so they cannot simply rebuild. Same mechanics as
# abm_expel_trade_offices: vanilla's disband_kontor pattern (peace_treaties/disband_kontor.txt) -
# `location_building_level = { building_type = ... owner = scope:recipient }` to find one specific
# country's building in a location, then `destroy_building_forcefully` to remove a foreign one.
#
# Unilateral (modelled on worsen_opinion), costs one diplomat, and PLAYER-ONLY in both directions
# so AI holdings stay exactly as vanilla.
#
# Consequences: the VICTIM gets vanilla's cb_destroyed_building against the actor for 10 years, and
# the ACTOR wears the abm_expelled_foreign_merchants country modifier for 10 years (trade income,
# merchant capacity and trade range each -10%). Repeated use extends both.

abm_expel_fondacos = {
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS

	potential = {
		scope:actor = {
			is_ai = no
			has_limited_diplomacy = no
		}
		scope:recipient = {
			is_ai = no
		}
	}

	allow = {
		scope:actor != scope:recipient
		scope:actor = {
			num_of_diplomats >= 1
		}
		# Kept in allow rather than potential so the entry greys out with a reason instead of the
		# action vanishing for every country that holds nothing here.
		custom_tooltip = {
			text = abm_expel_fondacos_has_fondacos_tt
			scope:actor = {
				any_owned_location = {
					OR = {
						location_building_level = {
							building_type = building_type:fondaco
							value > 0
							owner = scope:recipient
						}
						location_building_level = {
							building_type = building_type:hanseatic_kontor
							value > 0
							owner = scope:recipient
						}
						location_building_level = {
							building_type = building_type:abm_merchant_enclave
							value > 0
							owner = scope:recipient
						}
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "abm_expel_fondacos_SELECT_TITLE"
		none_available_msg_key = "abm_expel_fondacos_no_countries"
		column = {
			data = name
		}
		visible = {
			this != scope:actor
			within_diplomatic_range = scope:actor
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:recipient
			}
			scope:actor = {
				# Charged inside the guard so a cancelled selection is not billed.
				add_diplomats = -1
				every_owned_location = {
					if = {
						limit = {
							location_building_level = {
								building_type = building_type:fondaco
								value > 0
								owner = scope:recipient
							}
						}
						destroy_building_forcefully = "building(building_type:fondaco|scope:recipient)"
					}
					if = {
						limit = {
							location_building_level = {
								building_type = building_type:hanseatic_kontor
								value > 0
								owner = scope:recipient
							}
						}
						destroy_building_forcefully = "building(building_type:hanseatic_kontor|scope:recipient)"
					}
					if = {
						limit = {
							location_building_level = {
								building_type = building_type:abm_merchant_enclave
								value > 0
								owner = scope:recipient
							}
						}
						destroy_building_forcefully = "building(building_type:abm_merchant_enclave|scope:recipient)"
					}
				}
				# The rights go with the buildings, otherwise the target just rebuilds next month.
				if = {
					limit = {
						giving_scripted_relation = {
							type = relation_type:fondaco_rights
							target = scope:recipient
						}
					}
					remove_relation = {
						first = scope:actor
						second = scope:recipient
						type = relation_type:fondaco_rights
					}
				}

				# Our own merchants suffer for a while after throwing theirs out.
				add_country_modifier = {
					modifier = abm_expelled_foreign_merchants
					years = 10
					mode = add_and_extend
				}
			}
			# The wronged country may take up arms over it.
			scope:recipient = {
				add_casus_belli = {
					target = scope:actor
					type = casus_belli:cb_destroyed_building
					years = 10
				}
			}
		}
	}

	ai_tick = never
	ai_will_do = {
		add = -1000
	}
}

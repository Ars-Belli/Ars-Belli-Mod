pay_for_trade_access = {
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS

	potential = {
		scope:actor = {
			has_markets = no
		}
	}

	allow = {
		scope:actor != scope:recipient
		scope:actor = {
			gold > 0
		}
		scope:recipient = {
			has_markets = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			this != scope:actor
			within_diplomatic_range = scope:actor
		}
	}

	select_trigger = {
		looking_for_a = value
		target_flag = target_value
		name = "pay_for_access_select_gold_amount"
		min = 0
		max = scope:actor.gold
		step = scope:recipient.monthly_income_total
		default = {
			value = scope:recipient.monthly_income_total
			multiply = 12
		}
		ai_override_value = 0
	}

	effect = {
		if = {
			limit = { exists = scope:target_value }
			scope:actor = {
				add_gold = {
					value = scope:target_value
					multiply = -1
				}
			}
			scope:recipient = {
				add_gold = scope:target_value
			}
		}
		create_relation = {
			first = scope:recipient
			second = scope:actor
			type = relation_type:trade_access
		}
	}

	block_when_at_war = yes

	ai_tick = never
	ai_will_do = {
		add = -1000
	}

	diplo_chance = {
		base = -20
	}

	accept = {
		if = {
			limit = { exists = scope:target_value }
			add = {
				desc = "OFFERED_GOLD"
				value = scope:target_value
				divide = scope:recipient.monthly_income_total
				multiply = 8
			}
		}
		add = {
			desc = "game_concept_opinion"
			value = "scope:recipient.opinion(scope:actor)"
			divide = 5
		}
		if = {
			limit = {
				scope:recipient = { is_rival_of = scope:actor }
			}
			subtract = {
				desc = "game_concept_rival"
				value = 200
			}
		}
	}
}

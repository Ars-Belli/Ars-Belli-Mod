# ARS BELLI: starts the ars_belli_economic_support relation.
#
# This exists as an interaction rather than the relation's own offer flow for two
# reasons. A relation's select_trigger is the only selection stage in its flow, so a
# value-type one leaves the picker with no attribute columns and the game asserts on
# (AttributeColumns.GetSize() > 0) and crashes; here the country stage supplies columns
# first. And an interaction gives the recipient a real accept/decline popup, which is
# requirement Nr.3.

ars_belli_send_economic_support = {
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS

	potential = {
		scope:actor = {
			has_limited_diplomacy = no
		}
	}

	allow = {
		# One patron at a time, so the amount stored on the recipient is unambiguous.
		custom_tooltip = {
			text = ars_belli_economic_support_one_patron_tt
			NOT = {
				any_country = {
					giving_scripted_relation = {
						type = relation_type:ars_belli_economic_support
						target = scope:recipient
					}
				}
			}
		}
		custom_tooltip = {
			text = ars_belli_economic_support_points_tt
			scope:actor = {
				NOT = { is_over_defensive_limit = 1 }
			}
		}
		custom_tooltip = {
			text = ars_belli_economic_support_recipient_points_tt
			scope:recipient = {
				NOT = { is_over_defensive_limit = 1 }
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			within_diplomatic_range = scope:actor
			this != scope:actor
		}
	}

	select_trigger = {
		looking_for_a = value
		target_flag = target_value
		name = "ars_belli_economic_support_select_amount"
		min = 0
		max = scope:actor.monthly_income_total
		step = 1
		default = ars_belli_economic_support_default_amount
		ai_override_value = ars_belli_economic_support_default_amount
	}

	effect = {
		if = {
			limit = { exists = scope:target_value }
			scope:recipient = {
				set_variable = {
					name = ab_ecosupport_amount
					value = scope:target_value
				}
			}
			create_relation = {
				first = scope:actor
				second = scope:recipient
				type = relation_type:ars_belli_economic_support
			}
		}
	}

	block_when_at_war = no

	ai_tick = never
	ai_will_do = {
		add = -1000
	}

	diplo_chance = {
		base = 1000
	}

	# Free gold, so the AI takes it unless doing so would put it over its own limits.
	accept = {
		add = {
			desc = "BASE_VALUE"
			value = 1000
		}
		if = {
			limit = {
				scope:recipient = { is_over_defensive_limit = 1 }
			}
			subtract = {
				desc = ars_belli_economic_support_recipient_points_tt
				value = 5000
			}
		}
		if = {
			limit = {
				scope:recipient = { is_over_alliance_limit = 1 }
			}
			subtract = {
				desc = ars_belli_economic_support_recipient_points_tt
				value = 5000
			}
		}
	}
}

# Ars Belli - governor offices.
#
# A nobles building that converts an unused local governor post into government
# capacity. No levels: max_levels = 1, and since the engine allows only one instance of
# a building type per location that also makes each type unique per location. There is
# no country-wide limit, so one copy may stand in every eligible location as long as a
# free governor post remains (see abm_has_unused_local_governor) - each copy eats one.
# The reward lives in auto_modifiers/abm_country.txt and scales with the NUMBER of
# completed offices: a building modifier cannot be shown outside the capital
# (capital_country_modifier only applies "if built in the capital"), and these may
# stand elsewhere in the capital's area.
#
# Abolishing the office is allowed but expensive (on_destroyed): the governor post
# comes back, so the only reason to demolish is to dodge the trade - it costs gold,
# stability and political influence. Reforms already enacted with the extra slots
# stay enacted, hence the penalty.
#
# Shapes copied from vanilla: royal_court / chancery (capital_buildings.txt) for the
# location_potential shell, local_governor for forbidden_for_estates.

abm_provincial_chancellery = {
	audio_tier = 3
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	forbidden_for_estates = yes

	category = government_category
	employment_size = large_noble_employment

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_capital_build_time

	location_potential = {
		OR = {
			is_capital = yes
			AND = {
				is_city = yes
				is_core_of = owner
				area = owner.capital.area
			}
		}
	}

	allow = {
		OR = {
			is_capital = yes
			AND = {
				is_city = yes
				is_core_of = owner
				area = owner.capital.area
			}
		}
		scope:actor = {
			abm_has_unused_local_governor = yes
		}
	}

	remove_if = {
		location = {
			OR = {
				NOT = { is_city = yes }
				NOT = { is_core_of = owner }
			}
		}
	}

	on_destroyed = {
		location = {
			owner ?= {
				add_gold = {
					value = monthly_income_total
					multiply = -3
				}
				add_stability = -10
				add_political_influence = political_influence_severe_penalty
			}
		}
	}

	modifier = {
		pure_tooltip_entry = abm_governor_office_abolished_tt
	}

	possible_production_methods = {
		abm_provincial_chancellery_maintenance
	}

	ai_forbid_shutdown = yes
	important_for_AI = yes
}

# Naval counterpart of the Court Secretariat: converts an unused naval governor post
# into a cabinet seat. Placed where a naval governor may stand (port city, not the
# capital, not overseas) - connected ports included, per the naval_governor override
# in abm_naval_governors.txt.

abm_admiralty_board = {
	audio_tier = 3
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	forbidden_for_estates = yes

	category = government_category
	employment_size = large_noble_employment

	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_capital_build_time

	location_potential = {
		is_port = yes
		is_overseas_for_owner = no
		is_capital = no
		is_historical_capital = no
		has_proximity_source_building = no
	}

	allow = {
		is_port = yes
		is_overseas_for_owner = no
		is_capital = no
		is_historical_capital = no
		has_proximity_source_building = no
		scope:actor = {
			abm_has_unused_naval_governor = yes
		}
	}

	remove_if = {
		location = {
			OR = {
				NOT = { is_port = yes }
				is_capital = yes
			}
		}
	}

	on_destroyed = {
		location = {
			owner ?= {
				add_gold = {
					value = monthly_income_total
					multiply = -3
				}
				add_stability = -10
				add_political_influence = political_influence_severe_penalty
			}
		}
	}

	modifier = {
		pure_tooltip_entry = abm_governor_office_abolished_tt
	}

	possible_production_methods = {
		abm_admiralty_board_maintenance
	}

	ai_forbid_shutdown = yes
	important_for_AI = yes
}

# Ars Belli - Proconsulate.
#
# An annex to a local governor's seat: the governor's reach is extended by a naval
# governor post that is repurposed for civil administration. Three rules, in order:
#   - it can only stand where a local governor already stands (and it dies with him);
#   - it holds one naval governor post, counted exactly like a Provincial Administration
#     (see abm_has_unused_naval_governor - the post is subtracted, not a modifier);
#   - it amplifies the proximity source the location already has, +25% on the local
#     governor's 80, i.e. 100. raw_modifier, not modifier, so a badly supplied office
#     still gives the full quarter instead of a goods-access fraction of it.
#
# No levels and no country-wide limit: the engine allows one instance per location, and
# the local governor plus the naval post supply do the limiting.

abm_proconsulate = {
	audio_tier = 3
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	forbidden_for_estates = yes

	category = government_category
	employment_size = large_noble_employment

	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_capital_build_time

	location_potential = {
		has_building = building_type:local_governor
	}

	allow = {
		has_building = building_type:local_governor
		owner ?= scope:actor
		scope:actor = {
			abm_has_unused_naval_governor = yes
		}
	}

	# an annex to the governor's seat: no governor, no proconsulate
	remove_if = {
		location = {
			NOT = { has_building = building_type:local_governor }
		}
	}

	raw_modifier = {
		local_proximity_source_modifier = 0.25
	}

	possible_production_methods = {
		abm_proconsulate_maintenance
	}

	ai_forbid_shutdown = yes
	important_for_AI = yes
}

#####ArsBelliMod#####

# Ars Belli - smaller manpower buildings. Armory, Training Fields and their successors
# (Barracks, Regimental Camp, Conscription Center) come in half-size levels: half the manpower,
# half the soldier employment, half the maintenance and construction goods per level, and twice
# the level cap (abm_manpower_max_level in script_values/abm_building_caps.txt), so a location's
# cap grows in finer steps. Training Fields give 40% less again: 0.003 manpower a level
# (half of vanilla 0.01 is 0.005). Training slots per level (Barracks 1, Conscription Center 5)
# and the gold price stay vanilla.
#
# REPLACE: blocks are full copies of vanilla manpower_buildings.txt (game 1.4.9); every change is
# marked #ABM. Re-sync with vanilla on game updates. The 1337 start doubles its two armories.

REPLACE:armory = {
	audio_tier = 1
	is_foreign = no
	pop_type = soldiers
	max_levels = abm_manpower_max_level #ABM twice the levels, vanilla manpower_max_level
	
	category = military_category
	forbidden_for_estates = yes

	employment_size = abm_half_manpower_employment #ABM half, vanilla manpower_employment (1)
	
	town = yes
	city = yes
	megalopolis = yes
	
	important_for_AI = yes

	build_time = soldier_building
	
	obsolete = provincial_garrison
	obsolete = korean_barracks
	
	allow = {
		owner = { has_primary_or_accepted_culture = prev.dominant_culture }
	}
	remove_if = {
		NOT = { owner = { has_primary_or_accepted_culture = prev.location.dominant_culture } }
	}
	
	
	modifier = {
		local_manpower = 0.005 #ABM half, vanilla 0.01
		can_recruit_regiment_in_this_location = yes
	}
	
		
	unique_production_methods = {
		abm_armory_maintenance = { #ABM renamed from armory_maintenance so it does not collide with the vanilla production method
			icon_type = building_type
			icon = armory
			firearms = 0.25 #ABM half, vanilla 0.5
			leather = 0.25 #ABM half, vanilla 0.5
			cloth = 0.05 #ABM half, vanilla 0.10
			paper = 0.1 #ABM half, vanilla 0.20
			weaponry = 0.25 #ABM half, vanilla 0.5
			
			category = building_maintenance
		}

	}


	construction_demand = abm_half_early_soldier_building_construction #ABM half the goods, vanilla early_soldier_building_construction
}

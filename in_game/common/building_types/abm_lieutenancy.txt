# Ars Belli - lieutenancy, pointed at abm_has_unused_local_governor.
#
# lieutenancy is the non-DLC alternative to a local governor and shares its cap, so it has
# to see the same occupants (vanilla unique_buildings.txt). The ONLY change is the non-DLC
# cap check, replaced by the shared trigger; the DLC branch (modifier:num_lieutenancy) is
# untouched. Everything else is a verbatim copy - re-diff against vanilla after every game
# patch (expected delta: one allow line).

REPLACE:lieutenancy = {
	audio_tier = 2
	is_special = yes
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	forbidden_for_estates = yes

	category = government_category

	employment_size = normal_noble_employment

	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = government_build_time

	location_potential = {
		trigger_if = {
			limit = { exists = owner }
			is_capital = no
			is_historical_capital = no
			has_proximity_source_building = no
			trigger_if = {
					limit = { has_dlc = "d009_across_the_pillars" }
					OR = {
						AND = {
							is_in_scripted_geography = scripted_geography:castile_geography
							owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_castile_bureaucracy }
						}
						AND = {
							is_in_scripted_geography = scripted_geography:aragon_geography
							owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_aragon_bureaucracy }
						}
						AND = {
							is_in_scripted_geography = scripted_geography:portugal_geography
							owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_portugal_bureaucracy }
						}
						AND = {
							is_in_scripted_geography = scripted_geography:spanish_netherlands_geography
							owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_flanders_bureaucracy }
						}
						AND = {
							region = region:italy_region
							owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_italy_bureaucracy }
						}
					}
				}
				trigger_else = {
					has_road_to_capital = yes
				}
		}
		trigger_else = { always = no }
	}

	important_for_AI = yes
	AI_ignore_available_worker_flag = yes
	ai_ignore_maintenance = yes
	ai_forbid_shutdown = yes

	country_potential = {
		culture = { has_culture_group = culture_group:iberian_group }
		OR = {
			government_type ?= government_type:monarchy
			government_type ?= government_type:theocracy
			government_type ?= government_type:republic
		}
	}

	allow = {
		trigger_if = {
			limit = { has_dlc = "d009_across_the_pillars" }
			custom_tooltip = {
				text = lieutenancy_dlc_tt
				scope:actor = {
					total_building_levels_including_construction:lieutenancy < modifier:num_lieutenancy
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = lieutenancy_tt
				scope:actor = { abm_has_unused_local_governor = yes }
			}
		}
		owner = scope:actor
	}

	remove_if = {
		trigger_if = {
			limit = { has_dlc = "d009_across_the_pillars" }
			NOT = {
				OR = {
					AND = {
						location = { is_in_scripted_geography = scripted_geography:castile_geography }
						owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_castile_bureaucracy }
					}
					AND = {
						location = { is_in_scripted_geography = scripted_geography:aragon_geography }
						owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_aragon_bureaucracy }
					}
					AND = {
						location = { is_in_scripted_geography = scripted_geography:portugal_geography }
						owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_portugal_bureaucracy }
					}
					AND = {
						location = { is_in_scripted_geography = scripted_geography:spanish_netherlands_geography }
						owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_flanders_bureaucracy }
					}
					AND = {
						location = { region = region:italy_region }
						owner = { has_bureaucracy_of_type = bureaucracy_type:spa_council_of_italy_bureaucracy }
					}
				}
			}
		}
		trigger_else = {
			location = { has_road_to_capital = no }
			NOT = { owner = { at_war = yes } }
			NOT = { owner = { in_civil_war = yes } }
		}
	}

	modifier = {
		local_proximity_source = 75
		local_crown_estate_power = 1
	}

	unique_production_methods = {
		lieutenancy_maintenance = {
			icon_type = building_type
			icon = lieutenancy
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 2
			books = 0.5
			goods_gold = 1
			furniture = 1
			fine_cloth = 0.5

			category = building_maintenance
		}
	}
	construction_demand = local_governor_construction
}

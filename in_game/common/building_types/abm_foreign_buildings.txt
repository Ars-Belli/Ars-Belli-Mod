# ARS BELLI: the mod's versions of the two foreign trade buildings - trade_office and fondaco.
#
# ===========================================================================
# trade_office  (vanilla source: in_game/common/building_types/foreign_buildings.txt)
# Same fields as vanilla apart from:
#   * allow - a Trade Office built by a player in another PLAYER's land now needs that owner's
#     abm_trade_office_rights (see scripted_relations/abm_trade_office_rights.txt). AI builders
#     and AI-owned locations are exempt, so everything involving an AI stays vanilla.
#   * max_levels - points at abm_trade_office_max_level (script_values/abm_building_caps.txt),
#     a flat per-rank cap: town 1, city 2, megalopolis 3 (no development/population scaling).
#   * modifier / trade_office_requirements - Ars Belli rebalance: a few stronger offices instead
#     of many weak ones (per-level effect raised, upkeep inputs raised to match), plus
#     local_market_access like a marketplace, which is a LOCATION modifier so the location's owner
#     (the host that granted the rights) is the one who benefits from it.
#   * raw_modifier - local_monthly_prosperity, flat and unscaled by level (see the note on the
#     fondaco section below for why raw_modifier rather than modifier).
# ===========================================================================
REPLACE:trade_office = {
	audio_tier = 2
	is_foreign = yes
	stronger_power_projection = no
	max_levels = abm_trade_office_max_level
	need_good_relation = yes
	pop_type = burghers

	category = trade_category

	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	allow_wrong_startup = yes

	increase_per_level_cost = 1.0	#0

	country_potential = {
		NOR = {
			AND = {
				has_reform = government_reform:japanese_imperial_family
				country_type = building
			}
			has_reform = government_reform:japanese_clan
		}
	}

	allow = {
		market = { in_trade_range_of = scope:actor }
		trigger_if = {
			limit = {
				scope:actor = { is_ai = no }
				owner = { is_ai = no }
			}
			custom_tooltip = {
				text = abm_trade_office_rights_required_tt
				owner = {
					giving_scripted_relation = {
						type = relation_type:abm_trade_office_rights
						target = scope:actor
					}
				}
			}
		}
	}



	# A location that has fallen to a rural settlement may not host an office at
	# all (rural_settlement = no above). Demoting a city or megalopolis already
	# wipes the location's buildings, so this should never fire - it is a belt-and-
	# braces guard for a rural location that ends up holding a building anyway.
	remove_if = {
		location ?= { location_rank = location_rank:rural_settlement }
	}

	build_time = trade_post_build_time

	unique_production_methods = {
		trade_office_requirements = {
			icon_type = building_type
			icon = trade_office
			paper = 0.20							#0.1
			cloth = 0.20							#0.1
			salt = 0.20								#0.075
			category = building_maintenance
		}
	}


	modifier = {
		merchant_capacity_from_building = 1.0		#0.3
		building_owner_maritime_presence = 0.20 	#0.05
		merchant_power_from_building = 0.50			#0.2
		local_market_access = 0.03					#0.01			#marketplace value - location modifier, so the host (rights granter) benefits
	}

	raw_modifier = {
		local_monthly_prosperity = 0.001			#fixed regardless of level - raw_modifier is not scaled
	}
}

# ---------------------------------------------------------------------------
# fondaco  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# Same fields as vanilla apart from:
#   * max_levels - a flat 1, so a location holds a single fondaco level and it can never be
#     upgraded (vanilla read the SHARED guild_max_level, which 30 guild/marketplace buildings also
#     use, so it had to be replaced rather than retuned in place). Combined with the one-per-location
#     rule below, a location can only ever contain one fondaco, owned by one country.
#   * raw_modifier - local_monthly_development, flat +0.001 development/month to the location (and
#     so to the host). raw_modifier is not scaled by building level or goods access.
#   * allow - a location may hold only ONE fondaco, whoever owns it, so a second merchant republic
#     cannot build where one already stands (the owner filter still lets the holder upgrade).
# allow is untouched: vanilla already gates fondacos behind fondaco_rights.
# ---------------------------------------------------------------------------
REPLACE:fondaco = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes
	pop_type = burghers

	category = trade_category

	max_levels = 1
	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes

	increase_per_level_cost = 1.0	#0.5


	country_potential = {
		is_merchant_republic = yes
		NOT = { has_or_had_tag = HSA }
		# OR = {
		# 	AND = {
		# 		is_merchant_republic = yes
		#		culture = { has_culture_group = culture_group:italian_group }
		#	}
		#has_policy = ara_consulate_sea
		#}
	}
	allow = {
		within_naval_range_of = scope:actor
		owner = {
			giving_scripted_relation = {
				type = relation_type:fondaco_rights
				target = scope:actor
			}
		}
		# One trade post per location, whoever owns it: a second merchant republic cannot build here,
		# and a hanseatic_kontor occupies the same slot (see REPLACE:hanseatic_kontor below).
		custom_tooltip = {
			text = abm_fondaco_slot_taken_tt
			NOT = {
				any_foreign_buildings_in_location = {
					OR = {
						building_type = building_type:fondaco
						building_type = building_type:hanseatic_kontor
						building_type = building_type:abm_merchant_enclave
					}
					NOT = { owner = scope:actor }
				}
			}
		}
	}


	build_time = trade_post_build_time

	modifier = {
		merchant_power_from_building = 2.0				#1.0
		building_owner_maritime_presence = 0.1			#0.025
		merchant_capacity_from_building = 1				#0.2
		# merchants settle in the new location
		local_migration_attraction = 0.05				#0.025
	}

	raw_modifier = {
		local_monthly_development = 0.001				#fixed regardless of level - raw_modifier is not scaled
	}

	possible_production_methods = {
		fondaco_maintenance
	}
	construction_demand = fondaco_construction
}

# ---------------------------------------------------------------------------
# hanseatic_kontor  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# The Hansa's counterpart to the fondaco - HSA is excluded from fondacos, so this is its building.
# Same fields as vanilla apart from:
#   * max_levels - a flat 1 (vanilla scaled it off the rank: 3 base, +2 city, +10 megalopolis,
#     then x market_building_levels, i.e. 5-13 levels).
#   * allow - a new block (vanilla had none) enforcing the SAME one-per-location slot as fondacos:
#     a location holds a fondaco OR a kontor, never one of each. Shared with REPLACE:fondaco above.
# ---------------------------------------------------------------------------
REPLACE:hanseatic_kontor = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes
	pop_type = burghers
	
	category = trade_category
	
	max_levels = 1
	employment_size = large_foreign_trade_employment
	pop_size_created = large_foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	
	increase_per_level_cost = 0.5
	05
	country_potential = {
		modifier:allows_hanseatic_federation_buildings = yes
	}
	

	allow = {
		# Shared single slot with fondacos: a location takes one fondaco OR one kontor, never both.
		# The owner filter stops our own building from blocking anything of ours.
		custom_tooltip = {
			text = abm_fondaco_slot_taken_tt
			NOT = {
				any_foreign_buildings_in_location = {
					OR = {
						building_type = building_type:fondaco
						building_type = building_type:hanseatic_kontor
						building_type = building_type:abm_merchant_enclave
					}
					NOT = { owner = scope:actor }
				}
			}
		}
	}
	
	build_time = trade_post_build_time
	
	modifier = {
		merchant_power_from_building = 3.0			#1.0
		building_owner_maritime_presence = 0.2		#0.1
		merchant_capacity_from_building = 0.5		#0.25
	}
	
	raw_modifier = {
		local_monthly_development = 0.001			#fixed regardless of level - raw_modifier is not scaled
	}

	possible_production_methods = {
		kontor_maintenance
	}
	construction_demand = kontor_construction
}

# ---------------------------------------------------------------------------

	raw_modifier = {

	possible_production_methods = {
		fondaco_maintenance

# ARS BELLI: the mod's versions of the foreign buildings - trade_office, fondaco, the Hansa's three
# federation buildings (hanseatic_kontor, shipwright guild, town hall) and its burgtor.
#
# ===========================================================================
# trade_office  (vanilla source: in_game/common/building_types/foreign_buildings.txt)
# Same fields as vanilla apart from:
#   * allow - a Trade Office built by a player in another PLAYER's land now needs that owner's
#     abm_trade_office_rights (see scripted_relations/abm_trade_office_rights.txt). AI builders
#     and AI-owned locations are exempt, so everything involving an AI stays vanilla.
#   * max_levels - points at abm_trade_office_max_level (script_values/abm_building_caps.txt),
#     a flat per-rank cap: town 1, city 2, megalopolis 3 (no development/population scaling).
#   * modifier / trade_office_requirements - Ars Belli rebalance: a few stronger offices instead
#     of many weak ones (per-level effect raised, upkeep inputs raised to match), plus
#     local_market_access like a marketplace, which is a LOCATION modifier so the location's owner
#     (the host that granted the rights) is the one who benefits from it.
#   * raw_modifier - local_monthly_prosperity, flat and unscaled by level (see the note on the
#     fondaco section below for why raw_modifier rather than modifier).
# ===========================================================================
REPLACE:trade_office = {
	audio_tier = 2
	is_foreign = yes
	stronger_power_projection = no
	max_levels = abm_trade_office_max_level
	need_good_relation = yes
	pop_type = burghers

	category = trade_category

	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	allow_wrong_startup = yes

	increase_per_level_cost = 1.0	#0

	country_potential = {
		NOR = {
			AND = {
				has_reform = government_reform:japanese_imperial_family
				country_type = building
			}
			has_reform = government_reform:japanese_clan
		}
	}

	allow = {
		market = { in_trade_range_of = scope:actor }
		trigger_if = {
			limit = {
				scope:actor = { is_ai = no }
				owner = { is_ai = no }
			}
			custom_tooltip = {
				text = abm_trade_office_rights_required_tt
				owner = {
					giving_scripted_relation = {
						type = relation_type:abm_trade_office_rights
						target = scope:actor
					}
				}
			}
		}
	}



	# A location that has fallen to a rural settlement may not host an office at
	# all (rural_settlement = no above). Demoting a city or megalopolis already
	# wipes the location's buildings, so this should never fire - it is a belt-and-
	# braces guard for a rural location that ends up holding a building anyway.
	remove_if = {
		location ?= { location_rank = location_rank:rural_settlement }
	}

	build_time = trade_post_build_time

	unique_production_methods = {
		abm_trade_office_requirements = { #ABM renamed from trade_office_requirements so it does not collide with the vanilla production method
			icon_type = building_type
			icon = trade_office
			paper = 0.20							#0.1
			cloth = 0.20							#0.1
			salt = 0.20								#0.075
			category = building_maintenance
		}
	}


	modifier = {
		merchant_capacity_from_building = 1.0		#0.3
		building_owner_maritime_presence = 0.20 	#0.05
		merchant_power_from_building = 0.50			#0.2
		local_market_access = 0.03					#0.01			#marketplace value - location modifier, so the host (rights granter) benefits
	}

	raw_modifier = {
		local_monthly_prosperity = 0.001			#fixed regardless of level - raw_modifier is not scaled
	}
}

# ---------------------------------------------------------------------------
# fondaco  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# Same fields as vanilla apart from:
#   * max_levels - a flat 1, so a location holds a single fondaco level and it can never be
#     upgraded (vanilla read the SHARED guild_max_level, which 30 guild/marketplace buildings also
#     use, so it had to be replaced rather than retuned in place). Combined with the one-per-location
#     rule below, a location can only ever contain one fondaco, owned by one country.
#   * raw_modifier - local_monthly_development, flat +0.001 development/month to the location (and
#     so to the host). raw_modifier is not scaled by building level or goods access.
#   * allow - a location may hold only ONE fondaco, whoever owns it, so a second merchant republic
#     cannot build where one already stands (the owner filter still lets the holder upgrade).
#   * country_potential - vanilla's split restored and widened (2026-11-21): a Fondaco is the
#     trade post of ITALIAN and GREEK merchant republics. Every other merchant republic builds
#     abm_merchant_enclave instead, and the Hansa keeps its hanseatic_kontor.
# allow is untouched: vanilla already gates fondacos behind fondaco_rights.
# ---------------------------------------------------------------------------
REPLACE:fondaco = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes
	pop_type = burghers

	category = trade_category

	max_levels = 1
	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes

	increase_per_level_cost = 1.0	#0.5


	country_potential = {
		is_merchant_republic = yes
		culture = {
			OR = {
				has_culture_group = culture_group:italian_group
				has_culture_group = culture_group:greek_group
			}
		}
		NOT = { has_or_had_tag = HSA }
	}
	allow = {
		within_naval_range_of = scope:actor
		owner = {
			giving_scripted_relation = {
				type = relation_type:fondaco_rights
				target = scope:actor
			}
		}
		# One trade post per location, whoever owns it: a second merchant republic cannot build here,
		# and a hanseatic_kontor occupies the same slot (see REPLACE:hanseatic_kontor below).
		custom_tooltip = {
			text = abm_fondaco_slot_taken_tt
			NOT = {
				any_foreign_buildings_in_location = {
					OR = {
						building_type = building_type:fondaco
						building_type = building_type:hanseatic_kontor
						building_type = building_type:abm_merchant_enclave
					}
					NOT = { owner = scope:actor }
				}
			}
		}
	}


	build_time = trade_post_build_time

	modifier = {
		merchant_power_from_building = 2.0				#1.0
		building_owner_maritime_presence = 0.1			#0.025
		merchant_capacity_from_building = 1				#0.2
		# merchants settle in the new location
		local_migration_attraction = 0.05				#0.025
	}

	raw_modifier = {
		local_monthly_development = 0.001				#fixed regardless of level - raw_modifier is not scaled
	}

	possible_production_methods = {
		fondaco_maintenance
	}
	construction_demand = fondaco_construction
}

# ---------------------------------------------------------------------------
# hanseatic_kontor  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# The Hansa's counterpart to the fondaco - HSA is excluded from fondacos, so this is its building.
# Same fields as vanilla apart from:
#   * max_levels - a flat 1 (vanilla scaled it off the rank: 3 base, +2 city, +10 megalopolis,
#     then x market_building_levels, i.e. 5-13 levels).
#   * allow - a new block (vanilla had none) enforcing the SAME one-per-location slot as fondacos:
#     a location holds a fondaco OR a kontor, never one of each. Shared with REPLACE:fondaco above.
# ---------------------------------------------------------------------------
REPLACE:hanseatic_kontor = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes
	pop_type = burghers
	
	category = trade_category
	
	max_levels = 1
	employment_size = large_foreign_trade_employment
	pop_size_created = large_foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	
	increase_per_level_cost = 0.5
	country_potential = {
		modifier:allows_hanseatic_federation_buildings = yes
	}
	

	allow = {
		# Shared single slot with fondacos: a location takes one fondaco OR one kontor, never both.
		# The owner filter stops our own building from blocking anything of ours.
		custom_tooltip = {
			text = abm_fondaco_slot_taken_tt
			NOT = {
				any_foreign_buildings_in_location = {
					OR = {
						building_type = building_type:fondaco
						building_type = building_type:hanseatic_kontor
						building_type = building_type:abm_merchant_enclave
					}
					NOT = { owner = scope:actor }
				}
			}
		}
	}
	
	build_time = trade_post_build_time
	
	modifier = {
		merchant_power_from_building = 3.0			#1.0
		building_owner_maritime_presence = 0.2		#0.1
		merchant_capacity_from_building = 0.5		#0.25
	}
	
	raw_modifier = {
		local_monthly_development = 0.001			#fixed regardless of level - raw_modifier is not scaled
	}

	possible_production_methods = {
		kontor_maintenance
	}
	construction_demand = kontor_construction
}

# ---------------------------------------------------------------------------
# hanseatic_shipwright_guild  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# A Hansa-network building: buildable ONLY in the ports of the Hansa's own subjects, never in HSA's
# own land. Vanilla's location_potential already said that; what was missing is the cleanup -
# vanilla had no remove_if, so a guild survived its host being annexed or changing hands as a dead
# building. The guild is now destroyed as soon as the port is no longer owned by an HSA subject.
# (HSA has no subjects at start, so the two vanilla setup guilds in lubeck and hamburg are
# commented out in main_menu/setup/1337/07_cities_and_buildings.txt.)
# ---------------------------------------------------------------------------
REPLACE:hanseatic_shipwright_guild = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes

	pop_type = laborers
	max_levels = hanseatic_shipwright_guild_max_level
	category = military_category
	country_potential = {
		modifier:allows_hanseatic_federation_buildings = yes
	}
	location_potential = {
		owner ?= { is_subject_of = c:HSA }
		is_port = yes
	}
	employment_size = dock_employment
	pop_size_created = dock_employment
	# town = yes
	city = yes
	megalopolis = yes
	build_time = medium_port_building_time
	expensive = yes
	possible_production_methods = {
		port_building_maintenance
	}
	construction_demand = port_building_construction
	modifier = {
		max_ships_built_at_same_time = 1
		sailors_to_building_owner = 0.01
		can_build_ships_in_this_location = yes
		harbor_suitability = 0.1
		local_repair_speed = 0.01
	}
	remove_if = {
		OR = {
			location = { has_owner = no }
			location.owner ?= { NOT = { is_subject_of = c:HSA } }
		}
	}
}

# ---------------------------------------------------------------------------
# hanseatic_town_hall  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# Same fields as vanilla apart from:
#   * location_potential - vanilla hard-wired c:HSA.capital, i.e. the location variable, so the hall
#     could only ever stand in whatever location happens to BE the Hansa's capital. It is now
#     "the owner's own capital" (is_capital = yes), the same test vanilla uses for its capital-only
#     buildings.
#   * remove_if - vanilla destroyed the hall whenever its owner was NOT country_type = building, a
#     guard for the building-country Hansa which deleted the lubeck hall on the first tick. The hall
#     now dies with the federation reform, the same rule that drops the kontors.
# ---------------------------------------------------------------------------
REPLACE:hanseatic_town_hall = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes

	pop_type = nobles

	category = trade_category

	max_levels = 2
	employment_size = normal_noble_employment
	pop_size_created = normal_noble_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	increase_per_level_cost = 0.5

	expensive = yes

	location_potential = {
		is_capital = yes
	}

	country_potential = {
		modifier:allows_hanseatic_federation_buildings = yes
	}

	build_time = large_capital_build_time

	capital_country_modifier = {
		country_cabinet_efficiency = 0.05
		monthly_republican_tradition = 0.05
		parliament_base_support = 0.05
	}
	remove_if = {
		OR = {
			location = { has_owner = no }
			location.owner ?= {
				NOT = { modifier:allows_hanseatic_federation_buildings = yes }
			}
		}
	}

	possible_production_methods = {
		hanseatic_town_hall_maintenance
	}
	construction_demand = hanseatic_town_hall_construction
}

# ---------------------------------------------------------------------------
# hsa_burgtor  (vanilla source: in_game/common/building_types/unique_buildings.txt)
# The Hansa's city gate (Burgtor). The ONE Hansa building not gated on the federation reform - it
# comes from the advance hsa_hanseatic_fortifications, which only HSA can take. Same fields as
# vanilla apart from:
#   * max_levels - 1 instead of the vanilla 2, so a location holds a single gate.
#   * allow - a new block (vanilla had none): the location's OWNER must have granted us Fondaco
#     Rights, the same permission the mod's fondaco needs. Own locations are exempt (a country
#     cannot grant itself a relation, and the gate must be raisable in the Hansa's own cities).
#   * remove_if - vanilla's three clauses plus one: the gate is torn down when the host revokes the
#     rights (own locations exempt again). Vanilla also destroys it when the host's culture leaves
#     the Hansa's culture group, when the location loses its owner, or when the host is at war with
#     the building's owner.
# ---------------------------------------------------------------------------
REPLACE:hsa_burgtor = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	max_levels = 1
	pop_type = soldiers
	need_good_relation = yes
	category = defense_category
	employment_size = castle_employment
	pop_size_created = castle_employment
	city = yes
	megalopolis = yes
	build_time = small_fort_building
	expensive = yes
	possible_production_methods = {
		burgtor_maintenance
	}

	#root = location

	location_potential = {
		custom_tooltip = {
			text = location_owner_same_culture_group_as_us_tt
			owner ?= {
				culture = {
					any_culture_group = {
						any_culture_in_culture_group = {
							this = c:HSA.culture
						}
					}
				}
			}
		}
	}

	allow = {
		trigger_if = {
			limit = { NOT = { owner = scope:actor } }
			custom_tooltip = {
				text = abm_burgtor_rights_required_tt
				owner = {
					giving_scripted_relation = {
						type = relation_type:fondaco_rights
						target = scope:actor
					}
				}
			}
		}
	}

	#root = building

	remove_if = {
		OR = {
			location.owner ?= {
				culture = {
					any_culture_group = {
						NOT = {
							any_culture_in_culture_group = {
								this = c:HSA.culture
							}
						}
					}
				}
			}
			location = { has_owner = no }
			location.owner ?= {
				is_at_war_with = root.owner
			}
			location.owner ?= {
				NOT = {
					OR = {
						this = root.owner
						giving_scripted_relation = {
							type = relation_type:fondaco_rights
							target = root.owner
						}
					}
				}
			}
		}
	}

	#increase_per_level_cost = 1
	construction_demand = fort_construction
	modifier = {
		manpower_to_building_owner = 0.002
	}
	raw_modifier = {
		local_garrison_size = 0.1
		local_defensive = 0.1
		local_manpower_modifier = 0.05
	}
}

# ---------------------------------------------------------------------------
# abm_merchant_enclave  (NEW - no vanilla counterpart)
# A Merchant Enclave: the fondaco/kontor slot for every merchant government without a Fondaco -
# any merchant republic that is not Italian or Greek, plus the mod's own Chinese meritocratic
# government, Indian maritime league and bazaar republic, Bengalee meritocracy and trade emporium. Structurally identical to fondaco (same employment, upkeep PM, build time,
# one level, shares the one-per-location slot, gated on vanilla fondaco_rights) but with its
# modifier block at ~70% of the fondaco's (2026-11-21), so it is a weaker version of the same thing.
# ---------------------------------------------------------------------------
abm_merchant_enclave = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	need_good_relation = yes
	pop_type = burghers

	category = trade_category

	max_levels = 1
	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes

	increase_per_level_cost = 1.5

	country_potential = {
		OR = {
			# Ars Belli (2026-11-21): every merchant republic that is neither Italian nor Greek
			# uses the enclave where it used to fall through to a Fondaco. The Hansa keeps its
			# own hanseatic_kontor, and the Italian/Greek republics keep the Fondaco.
			AND = {
				is_merchant_republic = yes
				NOT = {
					culture = {
						OR = {
							has_culture_group = culture_group:italian_group
							has_culture_group = culture_group:greek_group
						}
					}
				}
				NOT = { has_or_had_tag = HSA }
			}
			has_reform = government_reform:abm_china_meritocratic_government
			has_reform = government_reform:abm_india_maritime_league
			has_reform = government_reform:abm_india_bazaar_republic
			has_reform = government_reform:abm_bengaleese_meritocracy
			has_reform = government_reform:abm_trade_emporium
			# Ars Belli: the Dutch Republic is a merchant government without an Italian/Greek
			# culture, so it gets the enclave rather than a Fondaco.
			has_reform = government_reform:dutch_republic_reform
		}
	}

	allow = {
		within_naval_range_of = scope:actor
		owner = {
			giving_scripted_relation = {
				type = relation_type:fondaco_rights
				target = scope:actor
			}
		}
		# Shared single slot with fondacos and kontors.
		custom_tooltip = {
			text = abm_fondaco_slot_taken_tt
			NOT = {
				any_foreign_buildings_in_location = {
					OR = {
						building_type = building_type:fondaco
						building_type = building_type:hanseatic_kontor
						building_type = building_type:abm_merchant_enclave
					}
					NOT = { owner = scope:actor }
				}
			}
		}
	}

	build_time = trade_post_build_time

	modifier = {
		merchant_power_from_building = 1.5
		building_owner_maritime_presence = 0.075
		merchant_capacity_from_building = 0.7
		# merchants settle in the new location
		local_migration_attraction = 0.05
	}

	raw_modifier = {
		local_monthly_development = 0.001
	}

	possible_production_methods = {
		fondaco_maintenance
	}
	construction_demand = fondaco_construction
}

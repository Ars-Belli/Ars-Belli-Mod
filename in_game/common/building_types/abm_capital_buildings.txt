REPLACE:order_headquarters = {
	audio_tier = 2 # 1.4 vanilla
	is_foreign = no
	pop_type = clergy
	max_levels = abm_starting_manpower_max_level
	
	category = government_category
	
	employment_size = 0.1
	
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes # 1.4 vanilla
	
	expensive = yes
	build_time = large_capital_build_time
	
	# 1.4 vanilla capital rules
	location_potential = {
		counts_as_capital = yes
	}

	allow = {
		is_capital = yes
		country_has_no_other_copy_of_building = { type = order_headquarters }
	}


	country_potential = {
		has_reform = government_reform:military_order_reform
		country_rank_level >= 2
	}
	
	remove_if = {
		OR = {
			is_no_longer_capital = yes
			location.owner ?= { 
				NOT = { 
					has_reform = government_reform:military_order_reform
				}
			}
		}
	}
	
	capital_modifier = {
		local_manpower = 0.020
		can_recruit_regiment_in_this_location = yes
	}
	
	raw_modifier = {
		local_defensive = 0.1		#fixed regardless of level - raw_modifier is not scaled
	}
	
	
	possible_production_methods = {
		new_order_headquarters_maintenance
	}

	graphical_tags = { palace }

	construction_demand = early_capital_building_construction
}

# Ars Belli 2026-11-19: vanilla gates the Royal Garden on
# `government_type ?= government_type:monarchy`; republics and theocracies may build it too.
# Everything else is verbatim vanilla (including the missing remove_if - vanilla has none, so
# the garden survives a government change).
REPLACE:royal_garden = {
	audio_tier = 2
	is_foreign = no
	pop_type = nobles
	max_levels = 1

	category = government_category

	employment_size = noble_cultural_employment

	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_capital_build_time

	location_potential = {
		counts_as_capital = yes
	}

	allow = {
		is_capital = yes
		country_has_no_other_copy_of_building = { type = royal_garden }
	}

	country_potential = {
		OR = {
			government_type ?= government_type:monarchy
			government_type ?= government_type:republic
			government_type ?= government_type:theocracy
		}
	}

	capital_modifier = {
	}

	capital_country_modifier = {
		diplomatic_capacity_modifier = 0.2
		cultural_influence_modifier = 0.05
		monthly_prestige = 0.05
	}

	unique_production_methods = {
		abm_royal_garden_maintenance = { #ABM renamed from royal_garden_maintenance so it does not collide with the vanilla production method
			icon_type = building_type
			icon = royal_garden
			furniture = 0.1
			tools = 0.1
			glass = 0.1
			fine_cloth = 0.05
			books = 0.1
			goods_gold = 0.05
			marble = 0.05

			category = building_maintenance
		}
	}
	construction_demand = prestige_building_construction
}

# Ars Belli: trade offices are a contested, shared resource per Location.
# A Location allows at most ab_trade_office_limit trade offices across ALL countries
# (game rule, 2-20; players compete for the limited slots). A country that already
# owns a trade office here can still upgrade its levels normally; only NEW offices
# are blocked once the slots are taken. The yearly pulse prunes any excess offices,
# removing the owners with the lowest market merchant power first.
REPLACE:trade_office = {
	is_foreign = yes
	stronger_power_projection = no
	max_levels = trade_office_max_level
	need_good_relation = yes
	pop_type = burghers

	category = trade_category

	employment_size = foreign_trade_employment

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	country_potential = {
		NOR = {
			AND = {
				has_reform = government_reform:japanese_imperial_family
				country_type = building
			}
			has_reform = government_reform:japanese_clan
		}
	}

	allow = {
		is_port = yes
		market = { in_trade_range_of = scope:actor }

		custom_tooltip = {
			text = ars_belli_trade_office_competition_tt
			OR = {
				# Upgrading a trade office we already own here is always allowed.
				any_foreign_buildings_in_location = {
					building_type = building_type:trade_office
					owner = scope:actor
				}
				# Otherwise a slot must be free: fewer offices than the configured limit
				# (game rule ab_trade_office_limit, 2-20) currently in this Location.
				NOT = {
					any_foreign_buildings_in_location = {
						count >= ab_trade_office_limit_value
						building_type = building_type:trade_office
					}
				}
			}
		}
	}



	build_time = trade_post_build_time

	unique_production_methods = {
		trade_office_requirements = {
			paper = 0.1
			category = building_maintenance
		}
	}


	modifier = {
		merchant_capacity_from_building = 0.3
		building_owner_maritime_presence = 0.05
		merchant_power_from_building = 0.2
	}
}

# ---------------------------------------------------------------------------
# Ars Belli: the formables of the north_german_region and south_german_region,
# as REPLACE: blocks so they can be edited here instead of in vanilla
# 00_formable_countries.txt. Bodies are verbatim vanilla except where noted; a
# comment line above each block carries the formable name as the game shows it.
# Formables of other regions live in abm_formables_<region>.txt.
# Blocks are sorted by rank, highest first: level 4 = empire, 3 = kingdom,
# 2 = duchy, 1 = county. Disabled (commented-out) blocks sort with their rank.
# ---------------------------------------------------------------------------

# Germany
REPLACE:GER_f = {

	level = 4
	required_locations_fraction = 0.70
	rule = plausible
	potential = {
		culture = { has_culture_group = culture_group:german_group }
		# Ars Belli: only a German power may claim this
		OR = {
			original_capital.region = region:north_german_region
			original_capital.region = region:south_german_region
		}
	}
	
	name = GER
	flag = GER
	adjective = GER_ADJ
	tag = GER
	color = map_GER
	
	regions = {
		north_german_region
		south_german_region
	}
	form_effect = {
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
	}
}

# Prussia
REPLACE:PRU_f = { #Prussia
	content_priority = 800
	level = 3
	required_locations_fraction = 0.7
	rule = historical

	capital_required = no

	potential = {
		# Ars Belli: Prussian, Brandenburgish or Pomeranian countries only
		OR = {
			culture = culture:prussian
			culture = culture:pruthenian # Ars Belli: the Teutonic Order's Baltic Prussians
			culture = culture:eastern_pomeranian
			culture = culture:brandenburgish
			culture = culture:western_pomeranian
		}
		# Ars Belli: and only from a capital in Brandenburg, Pomerania or Prussia
		OR = {
			original_capital.area = area:brandenburg_area
			original_capital.area = area:pomerania_area
			original_capital.area = area:prussia_area
		}
	}
	allow = {
		NOT = {
			has_reform = government_reform:military_order_reform
			has_estate_privilege = estate_privilege:clergy_powerful_dioceses
		}
		has_estate_privilege = estate_privilege:nobles_land_rights

		"estate_power(estate_type:clergy_estate)" < 0.20
		"estate_power(estate_type:nobles_estate)" > 0.20
		at_war = no

		# Ars Belli: the three thrones of Prussia in person
		owns = location:berlin
		owns = location:konigsberg
		owns = location:gdansk

		# Ars Belli: only a Protestant or an older heretic country may claim Prussia
		OR = {
			religion = religion:lutheran
			religion = religion:calvinist
			religion = religion:anglican
			religion = religion:lollardy
			religion = religion:catharism
			religion = religion:waldensian
		}
	}
	name = PRU
	flag = PRU
	adjective = PRU_ADJ
	tag = PRU
	color = map_PRU

	areas = {
		brandenburg_area
		pomerania_area
		prussia_area
	}

	form_effect = {
		if = {
			limit = {
				government_type != government_type:republic
				government_type != government_type:monarchy
			}
			change_government_type = government_type:monarchy
			add_legitimacy = legitimacy_radical_bonus
		}
		if = {
			limit = {
				NOT = { is_member_of_international_organization = international_organization:hre }
			}
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}

# Pomerania
# Ars Belli: tier 3 over the whole Wendish coast - Holstein, Mecklenburg and
# Pomerania - plus the Danzig province, and it requires Lubeck, Danzig and
# Stettin outright. The Danzig province sits in prussia_area/baltic_region, so
# this claim deliberately reaches outside the north German region.
REPLACE:POM_f = { #Pomerania
	required_locations_fraction = 0.6
	level = 3
	rule = historical
	potential = {
		OR = {
			culture = culture:eastern_pomeranian
			culture = culture:western_pomeranian
			culture = culture:hanseatic
			culture = culture:polabian # Ars Belli: the Polabian Slavs of the Pomeranian shore
		}
		# Ars Belli: and only from a capital in one of the required areas
		OR = {
			original_capital.area = area:holstein_area
			original_capital.area = area:mecklenburg_area
			original_capital.area = area:pomerania_area
		}
	}
	allow = {
		# Ars Belli: the three ports, held outright
		owns = location:lubeck
		owns = location:gdansk # the Danzig location is gdansk in EU5
		owns = location:stettin
	}

	name = POM
	flag = POM
	adjective = POM_ADJ
	tag = POM
	color = map_POM

	areas = {
		holstein_area
		mecklenburg_area
		pomerania_area
	}
	provinces = {
		danzig_province
	}
	form_effect = {
	}
}

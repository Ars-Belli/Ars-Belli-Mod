REPLACE:CHI_f = {
	level = 4
	required_locations_fraction = 0.75
	rule = historical

	potential = {
		culture = {
			has_culture_group = culture_group:chinese_group
		}
	}

	tag = CHI
	ars_belli_formable_countries
	regions = {
		west_china_region east_china_region north_china_region south_china_region
	}
	form_effect = {
		change_government_type = government_type:monarchy
		set_country_rank_effect = { rank = country_rank:rank_empire }
	}
}

ABSCP_f = {
	level = 3
	required_locations_fraction = 0.65
	rule = historical

	potential = {
		culture = {
			has_culture_group = culture_group:chinese_group
		}
	}

	allow = {
		has_culture_group = culture_group:chinese_group
		capital = { region = region:south_china_region }
		OR = {
			original_capital.region = region:north_china_region
			original_capital.region = region:east_china_region
			original_capital.region = region:west_china_region
			original_capital.region = region:south_china_region
		}
	}
	tag = ABSCP
	ars_belli_formable_countries
	regions = {
		south_china_region
	}
	form_effect = {
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
	}
}

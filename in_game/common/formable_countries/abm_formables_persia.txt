QAR_f = { #Qara Qoyunlu 
	level = 3
	required_locations_fraction = 0.7
	rule = historical

	potential = {
		government_type = government_type:steppe_horde
		OR = {
			culture = { has_culture_group = culture_group:iranian_group }
			culture = { has_culture_group = culture_group:mongolian_group }
		}
		capital = {
			OR = {
				region = region:persia_region
				region = region:crescent_region
				region = region:caucasus_region
			}
		}
	}

	allow = {
	}

	name = QAR
	flag = QAR
	adjective = QAR_ADJ
	tag = QAR
	color = map_QAR

	areas = {
		azerbaijan_area
		kordestan_area
		iraq_arabi_area
	}

	provinces = {
		tabriz_province
	}

	form_effect = {
		if = {
			limit = { NOT = { culture = culture:azeri_culture } }
			culture = { save_scope_as = qar_previous_culture }
			change_culture = culture:azeri_culture
			add_accepted_culture = scope:qar_previous_culture
		}
		if = {
			limit = { NOT = { religion = religion:shia } }
			change_religion = religion:shia
			set_religious_school = religious_school:isawiyya_school
			change_religion_for_ruler_and_family = {
				country = this
				religion = religion:shia
			}
		}
		if = {
			limit = {
				NOT = { capital = location:tabriz }
				location:tabriz = { owner = this }
			}
			set_capital = location:tabriz
			if = {
				limit = { location:tabriz = { NOT = { is_core_of = this } } }
				location:tabriz = { add_core = this }
			}
		}
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
	}
}


ABLOM_f = { # Lombardia
	level = 2
	required_locations_fraction = 0.60
	rule = historical

	potential = {
		capital = { region = region:italy_region }
		culture = { has_culture_group = culture_group:italian_group }
		OR = {
			culture = culture:lombard
			culture = culture:emilian
		}
	}

	allow = {
		OR = {
			capital = { area = area:lombardy_area }
			capital = { area = area:emilia_area }
		}
		OR = {
			culture = culture:lombard
			culture = culture:emilian
		}
		NOT = {
			any_owned_location = {
				OR = {
					area = area:lombardy_area
					area = area:piemonte_area
					area = area:emilia_area
					area = area:liguria_area
				}
				NOT = { is_core_of = root }
			}
		}
	}

	name = ABLOM
	flag = ABLOM
	adjective = ABLOM_ADJ
	tag = ABLOM
	color = map_ABLOM

	areas = {
		lombardy_area
		piemonte_area
		emilia_area
		liguria_area
	}
	form_effect = {
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
	}
}

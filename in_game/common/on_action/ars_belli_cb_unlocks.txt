on_game_start = {
	on_actions = {
		ars_belli_unlock_default_cbs
	}
}

on_new_country_formed = {
	on_actions = {
		ars_belli_unlock_default_cbs_new_country
	}
}

ars_belli_unlock_default_cbs = {
	effect = {
		every_country = {
			unlock_casus_belli = cb_deus_vult
		}
	}
}

ars_belli_unlock_default_cbs_new_country = {
	effect = {
		unlock_casus_belli = cb_deus_vult
	}
}

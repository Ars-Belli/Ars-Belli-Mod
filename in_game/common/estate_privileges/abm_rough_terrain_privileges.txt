# Ars Belli - rough-terrain estate privileges (peasants).
#
# These two lived in government_reforms until 2026-11-19. An estate privilege is the
# better home: it costs no government reform slot, and advances can unlock privileges
# directly (unlock_estate_privilege, vanilla peasants_yeomanry / vij_tamil_trading_rights),
# so nothing about the "granted by a rough-terrain advance" design is lost.
#
# mountains_proximity_impact / plateau_proximity_impact / hills_proximity_impact only
# exist at country scope (main_menu/common/modifier_type_definitions), so each rough
# terrain gets its own location_modifier block holding the exact inverse of that
# terrain's own penalties from in_game/common/topography:
#   proximity:        local_distance_from_capital_speed_propagation = 1/speed - 1
#                     (mountains 0.50 -> +100%, hills 0.75 -> +33%, plateau 0.875 -> +14%)
#   development:      local_monthly_development_modifier = -penalty
#                     (mountains -0.50, hills -0.25, plateau -0.10)
#
# Shapes copied from vanilla:
# - peasants privilege with several location_modifier + potential_trigger blocks:
#   in_game/common/estate_privileges/peasants_estate.txt (lat_compromise_with_greek_populace)
# - advance-unlocked privilege: the advance key makes it available, the privilege's
#   potential repeats the same `has_advance` gate (vanilla clergy_estate.txt
#   clergy_literacy_rights + 0_age_of_traditions.txt unlock_estate_privilege).
#   `has_unlocked_estate_privilege_trigger` is NOT this: it reads the country variable
#   `unlocked_estate_privilege_<id>` that only `unlock_estate_privilege_effect` sets.
# Shared location conditions: in_game/common/scripted_triggers/abm_rough_terrain_triggers.txt
abm_homeland_proximity_privilege = {
	estate = peasants_estate

	potential = {
		OR = {
			has_advance = arm_mount_ararat
			has_advance = tre_pontic_trail_mastery
			has_advance = ira_persian_plateau
			has_advance = abm_che_wetland_adaptation
			has_advance = abm_hng_mountain_adaptation
			has_advance = hab_expert_mountaineers
			has_advance = albanian_prokletije
			has_advance = scottish_highland_tradition
			has_advance = drokpa
			has_advance = inc_vertical_economy
			has_advance = mor_atlas_strongholds
			has_advance = a_permanent_capital
			has_advance = npl_himalayan_homeland
		}
	}

	# Once granted, it cannot be taken back (vanilla burghers_estate/nobles_estate idiom).
	can_revoke = {
		always = no
	}

	location_modifier = {
		potential_trigger = {
			abm_rough_terrain_mountains_in_original_capital_region = yes
		}
		# Each value cancels the terrain's own penalty exactly: the proximity term is
		# 1 / terrain_speed - 1 (mountains 0.50, hills 0.75, plateau 0.875), the
		# development term is -1 x the terrain's local_monthly_development_modifier.
		local_distance_from_capital_speed_propagation = 1.0
		local_monthly_development_modifier = 0.5
	}

	location_modifier = {
		potential_trigger = {
			abm_rough_terrain_hills_in_original_capital_region = yes
		}
		local_distance_from_capital_speed_propagation = 0.333
		local_monthly_development_modifier = 0.25
	}

	location_modifier = {
		potential_trigger = {
			abm_rough_terrain_plateaus_in_original_capital_region = yes
		}
		local_distance_from_capital_speed_propagation = 0.143
		local_monthly_development_modifier = 0.10
	}
	country_modifier = {
		# Every vanilla privilege declares the estate power it grants; without it the game
		# logs "has no power set" (estate_privilege.cpp:577) and the privilege stays unavailable.
		global_peasants_estate_power = 0.2
		pure_tooltip_entry = abm_homeland_proximity_privilege_tt
	}
}

# One obligation = one instance of this IO type (NOT unique -> many can coexist).
#   leader  = creditor (receives the money / payee)
#   member  = debtor   (pays the money / payer)
# Per-obligation data stored as variables on the IO instance:
#   var:ab_monthly      = gold paid per month
#   var:ab_months_left  = months remaining; decremented each month, disband at 0
ab_obligation = {
	has_target = no
	# no `unique` -> multiple simultaneous instances allowed (like coalitions)
	has_leader_country = yes
	leader_type = country
	disband_if_no_leader = yes

	show_on_diplomatic_map = yes # SPIKE: yes so the panel is easy to open; real feature = no
	show_leave_message = no
	can_invite_countries = no
	subject_limited = no

	# obligations must not consume diplomatic capacity
	diplomatic_capacity_cost = {
		add = 0
	}

	# hide the generic "create IO" diplo action; we create via a country_interaction
	create_visible_trigger = { always = no }
	invite_visible_trigger = { always = no }
	join_visible_trigger = { always = no }
	can_join_trigger = { always = yes }
	# Members CANNOT leave on their own — a debtor must not be able to walk away
	# from the debt. Obligations end only via Forgive (creditor) or expiry.
	can_leave_trigger = { always = no }
	can_lead_trigger = { always = yes }

	payments_implemented = {
		ab_obligation_payment
	}

	monthly_effect = {
		change_variable = {
			name = ab_months_left
			add = -1
		}
	}

	# Disband when the obligation runs out. Vanilla disbands IOs via this trigger
	# (root = IO), NOT via destroy_international_organization inside monthly_effect.
	# Use < 0 (not <= 0): the engine checks disband BEFORE that month's payment, so
	# <= 0 would skip the final payment (1 month short). With < 0, months_left = 0 is
	# still an active paying month; it disbands the month after reaching 0.
	# Forgive sets ab_months_left = -1 to fire this immediately.
	auto_disband_trigger = {
		var:ab_months_left < 0
	}

	modifier = {}
	leader_modifier = {}
	non_leader_modifier = {}
}

#Order Knights#

REPLACE:a_order_knights = {
	use_religious_order_coa = yes # 1.4 vanilla
	copy_from = a_age_1_traditions_heavy_cavalry
	is_special = yes

	strength_damage_taken = -0.125 #was -0.25
	morale_damage_taken = -0.125 #was -0.25

	upgrades_to = a_order_knights_2

	buildable = yes

	country_potential = {
		OR = {
			modifier:can_recruit_crusader_regiments = yes
			culture = { has_culture_group = culture_group:iberian_group }
		}
	}

	location_potential = {
		OR = {
			has_building_with_at_least_one_level = order_stronghold
			has_building_with_at_least_one_level = order_headquarters
		}
	}

	combat = {
		grasslands = 0.10
		wetlands = -0.10
	}

	#root = location. Checks location_potential first so don't need that scripting in
	mercenaries_per_location = {
		pop_type = nobles
		multiply = 0.2
	}

	gfx_tags = { crusader_tag heavy_tag }
}

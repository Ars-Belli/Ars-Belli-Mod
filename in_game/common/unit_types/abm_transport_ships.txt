# Transport ship food_storage_per_strength tuned so each transport carries
# the same displayed food as its age-equivalent logistic auxiliary unit
# (300 / 700 / 1200 / 1800 / 2750 / 3600 across ages 1-6).
#
# food_storage_per_strength is additive on top of the template's value, and
# displayed food = food_storage_per_strength * max_strength. Transports have
# max_strength = 1.0 from the navy_transport category, so mod value =
# target_displayed - template_food.
# Template food per age: 10 / 20 / 40 / 80 / 160 / 320.

REPLACE:n_cog = {
	category = navy_transport
	copy_from = n_age_1_traditions_transport
	#root = location. Checks location_potential first so don't need that scripting in
	mercenaries_per_location = {
		pop_type = burghers
		multiply = 0.05
	}

	age = "age_1_traditions"

	upgrades_to = n_hulk

	food_storage_per_strength = 290	# +290 on top of template's 10 -> 300 displayed (matches camp_followers)
}

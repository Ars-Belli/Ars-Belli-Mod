# Cultural regiments of the Pontic steppe.
#
# Each culture furnishes two or three historical unit types. They are ordinary
# recruitable regiments, not levies and not estate private armies.
#
# Two of the ten lines are gated on something other than plain culture:
#   - the Cossack regiments need the Cossacks estate and Cossack pops in the
#     recruiting location (a service institution, not an ethnicity);
#   - the Ruthenian regiments accept three routes: Ruthenian primary culture,
#     accepted Ruthenians while seated in Ruthenia or the steppe, or being Kyiv
#     or Ruthenia outright.
# The other eight lines use the default cultural gate: primary culture, or an
# accepted culture our primary culture views at `positive` or better, and in
# either case an original capital inside the steppes region.
#
# `cultural_view` is a culture-scope trigger, hence the `culture = { ... }`
# wrapper; vanilla does the same in common/script_values/io_policy.txt
# (`root.culture ?= { cultural_view = { target = ... value >= kindred } }`).
#
# Archetypes by design intent:
#   horse archer   - wins the skirmish, loses the melee (morale damage, low strength)
#   melee horse    - impact, then fades (strength damage, extra morale damage taken)
#   defensive foot - the anvil (negative damage taken on both morale and strength)
#   offensive foot - hits hard, dies faster (positive damage done AND taken)
#   balanced foot  - no penalty, no standout bonus

# crimean cultural regiment, age_2_renaissance.
abm_crimean_horse_archers_2 = {
	copy_from = a_age_2_renaissance_light_cavalry
	category = army_light_cavalry
	is_special = yes
	buildable = yes
	upgrades_to = abm_crimean_horse_archers_3
	initiative = 2
	morale_damage_done = 0.15
	strength_damage_done = -0.10
	supply_weight = -0.25
	impact = {
		flatland = -0.10
		grasslands = -0.10
		sparse = -0.10
		hills = 0.10
		mountains = 0.10
		woods = 0.10
		forest = 0.10
	}
	combat = {
		flatland = 0.10
		grasslands = 0.10
		sparse = 0.10
		hills = -0.10
		mountains = -0.10
		woods = -0.10
		forest = -0.10
	}
	gfx_tags = { archer_tag light_tag }

	# Primary culture always qualifies; an accepted culture needs our primary
	# culture's view of it at `positive` (the +1 tier) or better.
	country_potential = {
		OR = {
			culture = culture:crimean
			AND = {
				has_accepted_culture = culture:crimean
				culture = {
					cultural_view = {
						target = culture:crimean
						value >= positive
					}
				}
			}
		}
		# ...and the realm must be steppe-rooted. original_capital, not capital,
		# so moving the capital into the steppe later does not unlock them.
		original_capital ?= {
			region = region:steppes_region
		}
	}
}

	is_special = yes
	buildable = yes
	upgrades_to = abm_cossack_cavalry_4

	combat_speed = 2

	strength_damage_done = 0.10
	morale_damage_taken = -0.10
	supply_weight = -0.25

	impact = {
		flatland = -0.10
		grasslands = -0.10
		sparse = -0.10
		hills = 0.10
		mountains = 0.10
	}
	combat = {
		flatland = 0.10
		grasslands = 0.10
		sparse = 0.10
		hills = -0.10
		mountains = -0.10
	}
	gfx_tags = { sword_tag light_tag }

	# Cossack regiments are an estate privilege, not a cultural one: any country
	# that has the Cossacks estate may raise them, whatever its own culture, and
	# they may only be raised where Cossacks actually live - a handful is enough,
	# they need not be the majority.
	country_potential = {
		country_has_estate = estate_type:cossacks_estate
	}

	location_potential = {
		any_pop = {
			culture = culture:cossack_culture
		}
	}
}

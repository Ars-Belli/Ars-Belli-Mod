REPLACE:num_locations_impact = {
	scales_with = {
		value = num_locations
		divide = 20
	}

	fort_limit = 1
}

# Ars Belli urbanisation limit - the size terms, one auto_modifier each so they can be
# retuned independently of each other and of the fort limit in num_locations_impact:
#   +1 town point per 20 locations, and
#   +1 town point per 250k people (total_population is in thousands, hence divide = 250).
# Both divisors are also spelled out in the AUTO_MODIFIER_NAME_* loc keys in
# abm_urbanisation_l_english.yml, so the breakdown states the rule - change them together.
abm_urbanisation_locations_impact = {
	scales_with = {
		value = num_locations
		divide = 20
	}

	abm_urbanisation_limit = 1
}

abm_urbanisation_population_impact = {
	scales_with = {
		value = total_population
		divide = 250
	}

	abm_urbanisation_limit = 1
}

	potential_trigger = {
		used_fort_limit_percentage > 1.0

	}

	scales_with = {
		value = used_fort_limit_percentage

	}

}

}

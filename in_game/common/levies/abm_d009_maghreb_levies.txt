# so this keeps Masmuda tribesmen on foot, as they were. Vanilla body from 06_tribal_levies.txt apart from the NOT.
REPLACE:levy_tribal_cavalry = {
	size = levy_generic_infantry_size		#yes, its infantry, as it scales

	allowed_pop_type = tribesmen

	allow = {
		location = {
			market = {
				OR = {
					is_produced_in_market = goods:horses
					is_traded_in_market = goods:horses
				}
			}
		}
		NOT = {
			AND = {
				has_dlc = "d009_across_the_pillars"
				culture = culture:masmuda
			}
		}
	}
	country_allow = {
		has_estate_privilege = estate_privilege:tribes_tribal_levies
		NOT = { government_type = government_type:steppe_horde }
	}

	unit = a_tribal_cavalry
}

namespace = flavor_ssg

# The Jo house can abandon its hereditary Yuan office and claim a northern Goryeo restoration.
flavor_ssg.1 = {
	type = country_event
	title = flavor_ssg.1.title
	desc = flavor_ssg.1.desc
	outcome = neutral
	fire_only_once = yes

	dynamic_historical_event = {
		tag = SSG
		from = 1337.1.1
		to = 1400.1.1
		monthly_chance = 100
	}

	trigger = {
		custom_tooltip = {
			text = ssg_jo_restoration_undecided_tt
			NOR = {
				has_variable = ssg_jo_restoration_path
				has_variable = ssg_jo_restoration_rejected
			}
		}
	}

	option = {
		name = flavor_ssg.1.a
		set_variable = ssg_jo_restoration_path
		if = {
			limit = {
				situation:ssg_hanyang_jo_restoration = {
					situation_is_active = no
					situation_has_ended = no
				}
			}
			activate_situation = situation:ssg_hanyang_jo_restoration
		}
		change_country_name = SSG_NORTHERN_GORYEO
		change_country_adjective = SSG_NORTHERN_GORYEO_ADJ
		change_country_flag = SSG_NORTHERN_GORYEO
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
		add_legitimacy = legitimacy_severe_penalty
		add_stability = stability_mild_penalty
		add_country_modifier = {
			modifier = ssg_contested_royal_claim
			years = -1
			mode = replace
		}
		ai_chance = { base = 10 }
	}

	option = {
		name = flavor_ssg.1.b
		historical_option = yes
		set_variable = ssg_jo_restoration_rejected
		add_legitimacy = legitimacy_mild_bonus
		ai_chance = { base = 90 }
	}
}

	desc = flavor_ssg.200.desc
	outcome = neutral

	fire_only_once = yes

	dynamic_historical_event = {
		tag = SSG
		from = 1340.1.1
		to = 1400.1.1
		monthly_chance = 100
	}

	trigger = {
		has_variable = ssg_jo_restoration_path
		disease_is_active = disease:bubonic_plague
		NOT = { has_variable = ssg_plague_preparations_set }

			limit = { has_variable = ssg_economy_stage_1 }
			add_country_modifier = { modifier = ssg_granary_quarantine_reserves years = 10 mode = replace }

			limit = { has_variable = ssg_research_stage_1 }

	title = flavor_ssg.202.title
	desc = flavor_ssg.202.desc
	outcome = negative
	image = "gfx/interface/illustrations/situation/black_death.dds"
	fire_only_once = yes

	dynamic_historical_event = {
		tag = SSG
		from = 1340.1.1
		to = 1420.1.1
		monthly_chance = 100
	}

	trigger = {
		has_variable = ssg_jo_restoration_path
		has_variable = ssg_plague_preparations_set
		country_has_disease = disease:bubonic_plague
		NOT = { has_variable = ssg_black_death_reached_realm }
	}

	immediate = {
		set_variable = ssg_black_death_reached_realm
	}

	option = {
		name = flavor_ssg.202.a
		add_gold = { value = monthly_income_total multiply = -4 }
		custom_tooltip = flavor_ssg.202.a.tt
		hidden_effect = {
			if = {
				limit = { has_variable = ssg_economy_stage_1 }
				every_owned_location = {
					limit = { disease_presence = { disease = disease:bubonic_plague value > 0 } }
					change_disease_presence = { disease = disease:bubonic_plague value = -0.15 }
				}
			}
			else = {
				every_owned_location = {
					limit = { disease_presence = { disease = disease:bubonic_plague value > 0 } }
					change_disease_presence = { disease = disease:bubonic_plague value = -0.05 }
				}
			}
		}
		add_stability = stability_weak_bonus
		ai_chance = { base = 70 }
	}

	option = {
		name = flavor_ssg.202.b
		add_country_modifier = { modifier = ssg_sheltered_royal_court years = 5 mode = replace }
		if = {
			limit = {
				owns = location:hwaju
				location:hwaju = { disease_presence = { disease = disease:bubonic_plague value > 0 } }
			}
			location:hwaju = {
				change_disease_presence = { disease = disease:bubonic_plague value = -0.25 }
			}
		}
		add_legitimacy = legitimacy_weak_bonus
		ai_chance = { base = 30 }
	}
}

flavor_ssg.203 = {
	type = country_event
	title = flavor_ssg.203.title
	desc = flavor_ssg.203.desc
	outcome = positive
	image = "gfx/interface/illustrations/situation/ssg_hanyang_jo_restoration.dds"
	fire_only_once = yes

	dynamic_historical_event = {
		tag = SSG
		from = 1347.1.1
		to = 1425.1.1
		monthly_chance = 100
	}

	trigger = {
		has_variable = ssg_jo_restoration_path
		has_variable = ssg_plague_preparations_set
		NOT = { has_variable = ssg_plague_outcome_recorded }
		situation:black_death = {
			situation_is_active = no
			situation_has_ended = yes
		}
	}

	immediate = {
		set_variable = ssg_plague_outcome_recorded
		remove_country_modifier = ssg_mountain_cordons
		remove_country_modifier = ssg_guarded_caravan_stations
		remove_country_modifier = ssg_granary_quarantine_reserves
		remove_country_modifier = ssg_court_remedy_network
		remove_country_modifier = ssg_village_cordon_network
		remove_country_modifier = ssg_temple_relief_network
		remove_country_modifier = ssg_sheltered_royal_court
	}

	option = {
		name = flavor_ssg.203.a
		trigger = {
			NOR = {
				has_variable = ssg_black_death_reached_realm
				has_variable = had_black_death
			}
		}
		add_prestige = prestige_severe_bonus
		add_legitimacy = legitimacy_mild_bonus
		add_stability = stability_mild_bonus
		ai_chance = { base = 100 }
	}

	option = {
		name = flavor_ssg.203.b
		trigger = {
			OR = {
				has_variable = ssg_black_death_reached_realm
				has_variable = had_black_death
			}
		}
		add_gold = { value = monthly_income_total multiply = -3 }
		add_country_modifier = { modifier = ssg_post_plague_restoration years = 10 mode = replace }
		add_stability = stability_weak_bonus
		ai_chance = { base = 100 }
	}
}

# Reunification changes presentation, not the internal tag, so existing KOR DHEs remain coherent.
flavor_ssg.50 = {
	type = country_event
	title = flavor_ssg.50.title
	desc = flavor_ssg.50.desc
	outcome = neutral
	fire_only_once = yes

	option = {
		name = flavor_ssg.50.a
		set_variable = ssg_reunited_goryeo
		if = {
			limit = {
				is_situation_active = situation:ssg_hanyang_jo_restoration
			}
			end_situation = situation:ssg_hanyang_jo_restoration
		}
		change_country_name = KOR
		change_country_adjective = KOR_ADJ
		change_country_flag = KOR
		hidden_effect = { change_country_color = map_KOR }
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
		remove_country_modifier = ssg_contested_royal_claim
		add_country_modifier = { modifier = ssg_reunification_burdens years = 20 mode = replace }
		custom_tooltip = {
			text = flavor_ssg.50.integration_tt
			region:korea_region = {
				every_location_in_region = {
					limit = { owner ?= root }
					change_control = control_mild_penalty
				}
			}
		}
		add_prestige = prestige_severe_bonus
		add_legitimacy = legitimacy_severe_bonus
	}
}

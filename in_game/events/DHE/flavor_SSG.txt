namespace = flavor_ssg

# The Jo house can abandon its hereditary Yuan office and claim a northern Goryeo restoration.
flavor_ssg.1 = {
	type = country_event
	title = flavor_ssg.1.title
	desc = flavor_ssg.1.desc
	outcome = neutral
	fire_only_once = yes

	dynamic_historical_event = {
		tag = SSG
		from = 1337.1.1
		to = 1400.1.1
		monthly_chance = 100
	}

	trigger = {
		custom_tooltip = {
			text = ssg_jo_restoration_undecided_tt
			NOR = {
				has_variable = ssg_jo_restoration_path
				has_variable = ssg_jo_restoration_rejected
			}
		}
	}

	option = {
		name = flavor_ssg.1.a
		set_variable = ssg_jo_restoration_path
		change_country_name = SSG_NORTHERN_GORYEO
		change_country_adjective = SSG_NORTHERN_GORYEO_ADJ
		change_country_flag = SSG_NORTHERN_GORYEO
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
		add_legitimacy = legitimacy_severe_penalty
		add_stability = stability_mild_penalty
		add_country_modifier = {
			modifier = ssg_contested_royal_claim
			years = -1
			mode = replace
		}
		ai_chance = { base = 10 }
	}

	option = {
		name = flavor_ssg.1.b
		historical_option = yes
		set_variable = ssg_jo_restoration_rejected
		add_legitimacy = legitimacy_mild_bonus
		ai_chance = { base = 90 }
	}
}

# Reunification changes presentation, not the internal tag, so existing KOR DHEs remain coherent.
flavor_ssg.50 = {
	type = country_event
	title = flavor_ssg.50.title
	desc = flavor_ssg.50.desc
	outcome = neutral
	fire_only_once = yes

	option = {
		name = flavor_ssg.50.a
		set_variable = ssg_reunited_goryeo
		change_country_name = KOR
		change_country_adjective = KOR_ADJ
		change_country_flag = KOR
		hidden_effect = { change_country_color = map_KOR }
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
		remove_country_modifier = ssg_contested_royal_claim
		add_country_modifier = { modifier = ssg_reunification_burdens years = 20 mode = replace }
		custom_tooltip = {
			text = flavor_ssg.50.integration_tt
			region:korea_region = {
				every_location_in_region = {
					limit = { owner ?= root }
					change_control = control_mild_penalty
				}
			}
		}
		add_prestige = prestige_severe_bonus
		add_legitimacy = legitimacy_severe_bonus
	}
}

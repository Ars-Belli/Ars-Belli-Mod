flavor_cas_rio_salado.1 = { #Río Salado Prelude
	hide_portraits = yes
	type = country_event
	title = flavor_cas_rio_salado.1.title
	desc = flavor_cas_rio_salado.1.desc
	outcome = neutral
	historical_info = flavor_cas_rio_salado.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = CAS
		from = 1337.1.1
		to = 1350.1.1
		monthly_chance = 2
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		c:MOR = {
			event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		}
		#event_illustration_estate_background_effect = { background = type:army }

		ruler = {
			save_scope_as = target_character
		}
		c:CAS = {
			save_scope_as = target_ally
		}
	}

	trigger = {
		religion = religion:catholic
		has_ruler = yes
		c:MOR = {
			owns = location:algeciras
		}
		c:POR = {
			NOR = {
				is_rival_of = c:CAS
				is_enemy_of = c:CAS
			}
		}
		NOT = {
			is_at_war_with = c:POR
			is_at_war_with = c:MOR
		}
	}

	option = {
		name = flavor_cas_rio_salado.1.a
		historical_option = yes

		c:MOR = {
			add_opinion = { target = root modifier = claimed_our_land }
			trigger_event_non_silently = flavor_cas_rio_salado.2
		}
		add_casus_belli =  { type = casus_belli:cb_conquer_province province = location:algeciras.province target = c:MOR }
		#####ArsBelliMod##### Portugal's ally-or-pay event (.4) removed: forced diplomacy on a third country
	}
	option = {
		name = flavor_cas_rio_salado.1.b
		
		add_prestige = prestige_severe_penalty
		c:MOR = {
			add_opinion = { target = root modifier = refused_claim }
		}
	}
}

flavor_cas_rio_salado.3 = { #Granadas response
	type = country_event
	title = flavor_cas_rio_salado_response.34.title
	desc = flavor_cas_rio_salado_response.34.desc
	outcome = neutral

	major = yes
	major_trigger = {
		OR = {
			has_or_had_tag = CAS
			has_or_had_tag = MOR
		}
	}
	
	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		#event_illustration_estate_background_effect = { background = type:army }

		c:MOR = {
			save_scope_as = target_ally
		}
		c:CAS = {
			save_scope_as = target_enemy
		}
	}

	option = {
		name = flavor_cas_rio_salado_response.34.a

		ai_will_select = {
			value = 200
			add = "opinion(c:MOR)"
		}

		if = {
			limit = { is_subject_of = c:CAS }
			c:CAS = {
				cancel_subject = root
				add_casus_belli = {
					target = root
					type = casus_belli:cb_conquer_province
					province = root.capital.province
				}
			}
			if = {
				limit = {
					has_dlc = "d009_across_the_pillars"
					is_situation_active = situation:battle_for_the_strait
					situation:battle_for_the_strait = {
						has_variable = influence_over_granada
						var:influence_over_granada < 0
					}
				}
				reset_influence_over_granada_effect = yes
			}
		}

		if = {
			limit = {
				NOT = {
					is_allied_with = { target = c:MOR }
				}
			}
			create_relation = {
				first = this
				second = c:MOR
				type = relation_type:alliance
			}
		}
		else_if = {
			limit = {
				is_allied_with = { target = c:MOR }
			}
			add_opinion = { target = c:MOR modifier = opinion_diplomatic_move }
		}
		if = {
			limit = {
				has_dlc = "d009_across_the_pillars"
				is_situation_active = situation:battle_for_the_strait
			}
			change_influence_over_granada_effect = { value = bfts_pull_toward_mor_mid }
			change_granadan_sovereignty_effect = { value = 5 }
		}
		
	}

	option = {
		name = flavor_cas_rio_salado_response.34.b

		ai_will_select = {
			value = 200
			add = "opinion(c:CAS)"
		}

		if = {
			limit = { is_subject_of = c:CAS }
			add_opinion_mutual_effect = {
				modifier = gra_maintained_tributary
				target = c:CAS
			}
		}

		if = {
			limit = {
				has_dlc = "d009_across_the_pillars"
				is_situation_active = situation:battle_for_the_strait
			}
			change_influence_over_granada_effect = { value = bfts_pull_toward_cas_mid }
			change_granadan_sovereignty_effect = { value = -10 }
		}
		else_if = {
			limit = { is_subject = no }
			add_prestige = prestige_extreme_penalty
			add_legitimacy = legitimacy_extreme_penalty
		}

	}
}

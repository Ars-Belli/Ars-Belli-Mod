namespace = abm_enforce_peace

# Step 2 of the Enforce Peace chain. The defender has already agreed to a white peace;
# this asks the attacker. Fired from common/country_interactions/abm_enforce_peace.txt
#
# The interaction cannot hand scopes over directly - anything saved with save_scope_as in
# a country_interaction effect comes back as "Undefined event target" here - so it passes
# the three participants as variables on this country and the immediate block below turns
# them back into scopes. Those scopes then carry normally into .2 and .3, which are fired
# from the options, because event-to-event propagation does work.
#
# root = attacker leader (this country)
abm_enforce_peace.1 = {
	type = country_event
	title = abm_enforce_peace.1.title
	desc = abm_enforce_peace.1.desc
	outcome = neutral

	immediate = {
		var:abm_ep_enforcer = { save_scope_as = ep_enforcer }
		var:abm_ep_defender = { save_scope_as = ep_defender }
		var:abm_ep_war = { save_scope_as = ep_war }
	}

	# Accept the white peace.
	option = {
		name = abm_enforce_peace.1.a
		white_peace = scope:ep_war

		scope:ep_enforcer = {
			trigger_event_silently = abm_enforce_peace.2
		}

		remove_variable = abm_ep_enforcer
		remove_variable = abm_ep_defender
		remove_variable = abm_ep_war

		ai_chance = {
			factor = 2
			# A free exit looks far better when the war is going badly.
			modifier = {
				factor = 3
				war_score_in_war_whole_side = {
					war = scope:ep_war
					value < 0
				}
			}
			modifier = {
				factor = -1
				war_score_in_war_whole_side = {
					war = scope:ep_war
					value > 25
				}
			}
		}
	}

	# Refuse, and take the enforcer into the war against us.
	option = {
		name = abm_enforce_peace.1.b

		scope:ep_enforcer = {
			join_war_with = {
				war = scope:ep_war
				target = scope:ep_defender
				call_in_subjects = yes
				notify_opposing_side = yes
			}
			trigger_event_silently = abm_enforce_peace.3
		}

		remove_variable = abm_ep_enforcer
		remove_variable = abm_ep_defender
		remove_variable = abm_ep_war

		ai_chance = {
			factor = 2
			modifier = {
				factor = -1
				war_score_in_war_whole_side = {
					war = scope:ep_war
					value < -25
				}
			}
			modifier = {
				factor = 3
				war_score_in_war_whole_side = {
					war = scope:ep_war
					value > 25
				}
			}
		}
	}
}

# Notification to the enforcer: the attacker took the white peace.
abm_enforce_peace.2 = {
	type = country_event
	title = abm_enforce_peace.2.title
	desc = abm_enforce_peace.2.desc
	outcome = positive

	option = {
		name = abm_enforce_peace.2.a
	}
}

# Notification to the enforcer: the attacker refused, so we are now in the war.
abm_enforce_peace.3 = {
	type = country_event
	title = abm_enforce_peace.3.title
	desc = abm_enforce_peace.3.desc
	outcome = negative

	option = {
		name = abm_enforce_peace.3.a
	}
}

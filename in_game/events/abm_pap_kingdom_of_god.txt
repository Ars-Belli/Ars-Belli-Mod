namespace = abm_pap_kog

#The Kingdom of God on Earth
abm_pap_kog.1 = {
	type = country_event
	fire_only_once = yes
	title = abm_pap_kog.1.title
	desc = abm_pap_kog.1.desc
	outcome = neutral

	immediate = {
		ruler ?= {
			save_scope_as = target_character
		}
	}

	dynamic_historical_event = {
		tag = PAP
		from = 1300.1.1
		to = 1837.1.1
		monthly_chance = 100
	}

	trigger = {
		has_ruler = yes
		has_reform = government_reform:papacy_reform
		region:italy_region = {
			any_location_in_region = {
				OR = {
					is_ownable = no
					top_owner ?= root
				}
				count >= 200
			}
		}
	}

	option = {
		name = abm_pap_kog.1.a
		trigger = {
			not = { has_reform = government_reform:kingdom_of_god_reform }
		}

		unlock_government_reform_effect = { type = kingdom_of_god_reform }
		#1.4: dropped the kingdom_of_god_slot_modifier hack (the modifier no longer exists; add_reform
		#forces the reform in on its own, exactly like vanilla's flavor_pap.20)
		add_reform = government_reform:kingdom_of_god_reform
        add_government_power = government_power_severe_bonus

	}

	option = {
		name = abm_pap_kog.1.b
		trigger = {
			not = { has_reform = government_reform:kingdom_of_god_reform }
		}

		unlock_government_reform_effect = { type = kingdom_of_god_reform }
		#1.4: see option a - the slot modifier is gone
		add_reform = government_reform:kingdom_of_god_reform
		add_prestige = prestige_severe_bonus
	}
}
